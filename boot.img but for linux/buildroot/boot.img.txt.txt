Run

sudo apt install android-tools-mkbootimg
cp /path/to/your/compiled/kernel/arch/arm64/boot/Image ~/ums512_kernel/abootimg
cp /ums512_kernel/buildroot/output/images ~/ums512_kernel/abootimg

cd /ums512_kernel/abootimg
mkbootimg \
  --kernel kernelImage \
  --ramdisk rootfs.cpio.gz \
  --base 0x00000000 \
  --kernel_offset 0x00008000 \
  --ramdisk_offset 0x05400000 \
  --tags_offset 0x00000100 \
  --pagesize 2048 \
  --cmdline "console=ttyS1,115200n8 root=/dev/block/by-name/super rootfstype=ext4 rw init=/sbin/init loglevel=7" \
  -o boot_linux.img
