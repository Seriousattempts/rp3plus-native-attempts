#!/system/bin/sh

# =========================================================================
# Debian Trixie Comprehensive Hardware Test Script
# Device: Retroid Pocket 3 Plus (UMS512 / Mali-G52)
# Version: 5.0 - Fixed Debian Paths & Mesa Installation
# =========================================================================

# === PARTITION CONFIGURATION ===
SUPER_PARTITION="/dev/block/mmcblk0p50"
DISTRO_PARTITION="/dev/block/mmcblk0p77"
SOCKO_PARTITION="/dev/block/mmcblk0p52"
DISTRO_MOUNT="/mnt/distro"
DISTRO_ROOT="/mnt/distro/debian-chroot"
BOOT_PARTITION="/dev/block/mmcblk1p1"
SD_MOUNT=""
TEMP_DIR=""
VENDOR_MOUNT=""
VENDOR_IMG=""
VENDOR_LOOP=""
LOGFILE="/tmp/debian_v5_test.log"
TEST_OUTPUT=""
PACKAGE_LIST=""
DEBIAN_USER="p3plus"
MESA_TARBALL="mesa-for-android-container_25.3.0-devel-20250725_debian_arm64.tar.gz"

# === LOGGING SETUP ===
> "$LOGFILE"
log() {
    TIMESTAMP="$(date '+%Y-%m-%d %H:%M:%S' 2>/dev/null || echo 'TIME')"
    echo "[$TIMESTAMP] $*" | tee -a "$LOGFILE"
}

log_separator() {
    log "========================================================================="
}

log_subsection() {
    log "-------------------------------------------------------------------------"
}

log_success() {
    log "✓ SUCCESS: $*"
}

log_error() {
    log "✗ ERROR: $*"
}

log_warning() {
    log "⚠ WARNING: $*"
}

log_info() {
    log "ℹ INFO: $*"
}

log_attempt() {
    log "→ ATTEMPT: $*"
}

log_separator
log "=== Debian Trixie Comprehensive Hardware Test Script ==="
log "=== Version 5.0 - Fixed Debian Paths & Mesa Installation ==="
log_separator
log_info "Device: Retroid Pocket 3 Plus"
log_info "Platform: UNISOC UMS512 with Mali-G52 GPU"
log_info "Kernel: $(uname -r)"
log_info "Architecture: $(uname -m)"
log_info "Hostname: $(hostname)"
log_info "Test Date: $(date '+%A, %B %d, %Y at %H:%M:%S %Z')"
log ""

log_info "Version 5.0 Changes:"
log_info "  - Fixed PATH to use Debian binaries only (not Android vendor)"
log_info "  - Previous issue: /vendor/bin contaminating chroot PATH"
log_info "  - Added Mesa 25.3.0 installation from SD card tarball"
log_info "  - Added panfrost_icd.aarch64.json verification"
log_info "  - Excluded sprd_vdsp.ko module (per user request)"
log_info "  - Vendor partition used only for kernel module loading"
log ""

# === SD CARD MOUNT FOR PERSISTENT LOGGING ===
log_separator
log "=== PHASE 1: SD CARD MOUNT FOR PERSISTENT LOGGING ==="
log_separator

log_info "Checking if SD card is already mounted..."
if grep -q "$BOOT_PARTITION" /proc/mounts 2>/dev/null; then
    SD_MOUNT=$(grep "$BOOT_PARTITION" /proc/mounts | awk '{print $2}' | head -1)
    log_success "SD card already mounted at: $SD_MOUNT"
    log_info "Mount details: $(mount | grep $BOOT_PARTITION)"
else
    log_info "SD card not mounted, attempting mount..."
    MOUNT_ATTEMPTS=0
    
    for mp in /sdcard /boot /mnt/boot /mnt/sdcard; do
        MOUNT_ATTEMPTS=$((MOUNT_ATTEMPTS + 1))
        log_attempt "Attempt $MOUNT_ATTEMPTS: Trying mount point $mp..."
        
        mkdir -p "$mp" 2>/dev/null
        if [ ! -d "$mp" ]; then
            log_warning "Failed to create directory $mp"
            continue
        fi
        
        log_info "Directory $mp created successfully"
        log_attempt "Attempting mount with filesystem type: exfat"
        
        if mount -t exfat "$BOOT_PARTITION" "$mp" 2>&1 | tee -a "$LOGFILE"; then
            SD_MOUNT="$mp"
            log_success "SD card successfully mounted at: $SD_MOUNT"
            log_info "Mount verification: $(mount | grep $BOOT_PARTITION)"
            break
        else
            log_warning "Mount attempt failed for $mp"
        fi
    done
    
    if [ -z "$SD_MOUNT" ]; then
        SD_MOUNT="/tmp/boot_fallback"
        mkdir -p "$SD_MOUNT"
        log_error "All mount attempts failed"
        log_warning "Using fallback location: $SD_MOUNT"
        log_warning "Logs will NOT persist after reboot - stored in RAM only"
    fi
fi

log ""
log_info "Creating logs directory structure..."
mkdir -p "$SD_MOUNT/logs" 2>/dev/null

if [ -d "$SD_MOUNT/logs" ]; then
    log_success "Logs directory created: $SD_MOUNT/logs"
    log_info "Directory permissions: $(ls -ld $SD_MOUNT/logs)"
else
    log_error "Failed to create logs directory"
fi

# Relocate log to SD card with timestamp
if [ "$SD_MOUNT" != "/tmp/boot_fallback" ] && [ -w "$SD_MOUNT/logs" ]; then
    NEW_LOG="${SD_MOUNT}/logs/debian_v5_$(date +%Y%m%d_%H%M%S).log"
    log_info "Attempting to relocate log file to: $NEW_LOG"
    
    if cat "$LOGFILE" > "$NEW_LOG" 2>/dev/null; then
        LOGFILE="$NEW_LOG"
        log_success "Log file successfully relocated to SD card"
        log_info "Active log file: $LOGFILE"
    else
        log_error "Failed to relocate log file, continuing with temporary location"
    fi
else
    log_warning "SD card logs directory not writable"
    log_warning "Log remains in memory at: $LOGFILE"
fi

TEST_OUTPUT="${SD_MOUNT}/logs/debian_hardware_report_$(date +%Y%m%d_%H%M%S).txt"
PACKAGE_LIST="${SD_MOUNT}/logs/debian_packages_$(date +%Y%m%d_%H%M%S).txt"

log_info "Test output file: $TEST_OUTPUT"
log_info "Package list file: $PACKAGE_LIST"
log ""

# === WAIT FOR BLOCK DEVICES ===
log_separator
log "=== PHASE 2: BLOCK DEVICE DETECTION ==="
log_separator

log_info "Waiting for super partition: $SUPER_PARTITION"
WAIT=0
WAIT_MAX=50

while [ ! -b "$SUPER_PARTITION" ] && [ $WAIT -lt $WAIT_MAX ]; do
    sleep 0.1
    WAIT=$((WAIT + 1))
    if [ $((WAIT % 10)) -eq 0 ]; then
        log_info "Still waiting... (${WAIT}/${WAIT_MAX})"
    fi
done

if [ ! -b "$SUPER_PARTITION" ]; then
    log_error "Super partition not found after ${WAIT_MAX} attempts"
    log_error "Device node: $SUPER_PARTITION"
    log_info "Available block devices in /dev/block/:"
    ls -la /dev/block/ | grep mmcblk0 | tee -a "$LOGFILE"
    log_info "Exiting to recovery..."
    sync
    exec /system/bin/realrecovery "$@"
fi

log_success "Super partition found: $SUPER_PARTITION"
log_info "Block device info: $(ls -l $SUPER_PARTITION)"

log ""
log_info "Waiting for distro partition: $DISTRO_PARTITION"
WAIT=0

while [ ! -b "$DISTRO_PARTITION" ] && [ $WAIT -lt $WAIT_MAX ]; do
    sleep 0.1
    WAIT=$((WAIT + 1))
    if [ $((WAIT % 10)) -eq 0 ]; then
        log_info "Still waiting... (${WAIT}/${WAIT_MAX})"
    fi
done

if [ ! -b "$DISTRO_PARTITION" ]; then
    log_error "Distro partition not found after ${WAIT_MAX} attempts"
    log_error "Device node: $DISTRO_PARTITION"
    log_info "Exiting to recovery..."
    sync
    exec /system/bin/realrecovery "$@"
fi

log_success "Distro partition found: $DISTRO_PARTITION"
log_info "Block device info: $(ls -l $DISTRO_PARTITION)"

log ""
log_info "Version 4.0+ Fix: Using corrected SOCKO partition path"
log_info "Previous attempts used: /dev/block/mmcblk0p49 (incorrect)"
log_info "Correct SOCKO partition: $SOCKO_PARTITION"
if [ ! -b "$SOCKO_PARTITION" ]; then
    log_warning "SOCKO partition not found: $SOCKO_PARTITION"
    log_warning "Hardware module loading may be limited"
else
    log_success "SOCKO partition found: $SOCKO_PARTITION"
    log_info "Block device info: $(ls -l $SOCKO_PARTITION)"
fi
log ""

# === MOUNT DISTRO PARTITION ===
log_separator
log "=== PHASE 3: DISTRO PARTITION MOUNTING ==="
log_separator

log_info "Creating mount point: $DISTRO_MOUNT"
mkdir -p "$DISTRO_MOUNT" 2>/dev/null

if [ -d "$DISTRO_MOUNT" ]; then
    log_success "Mount point created: $DISTRO_MOUNT"
else
    log_error "Failed to create mount point: $DISTRO_MOUNT"
    log_info "Exiting to recovery..."
    sync
    exec /system/bin/realrecovery "$@"
fi

log_info "Attempting to mount distro partition..."
log_info "Source: $DISTRO_PARTITION"
log_info "Target: $DISTRO_MOUNT"

if mount "$DISTRO_PARTITION" "$DISTRO_MOUNT" 2>&1 | tee -a "$LOGFILE"; then
    log_success "Distro partition mounted successfully"
    
    MOUNT_INFO=$(mount | grep "$DISTRO_PARTITION")
    log_info "Mount details: $MOUNT_INFO"
    
    FS_TYPE=$(echo "$MOUNT_INFO" | awk '{print $5}')
    MOUNT_OPTS=$(echo "$MOUNT_INFO" | sed 's/.*(\(.*\))/\1/')
    
    log_info "Filesystem type: ${FS_TYPE:-Unknown}"
    log_info "Mount options: ${MOUNT_OPTS:-Unknown}"
    
    log ""
    log_info "Verifying mount accessibility..."
    if ls "$DISTRO_MOUNT" >/dev/null 2>&1; then
        log_success "Mount point is accessible"
    else
        log_error "Mount point is not accessible"
    fi
else
    log_error "Failed to mount distro partition"
    log_info "Checking partition filesystem..."
    blkid "$DISTRO_PARTITION" 2>&1 | tee -a "$LOGFILE"
    log_info "Exiting to recovery..."
    sync
    exec /system/bin/realrecovery "$@"
fi
log ""

# === INSPECT PARTITION 77 ===
log_separator
log "=== PHASE 4: PARTITION 77 INSPECTION (Debian Rootfs) ==="
log_separator

log_info "Gathering partition information..."
log ""
log_info "Mount point contents (top level):"
log_subsection
ls -lah "$DISTRO_MOUNT" 2>&1 | tee -a "$LOGFILE"
log_subsection

log ""
log_info "Disk usage statistics:"
df -h "$DISTRO_MOUNT" 2>&1 | tee -a "$LOGFILE"
log ""

# === VERIFY DEBIAN ROOTFS ===
log_separator
log "=== PHASE 5: DEBIAN ROOTFS VERIFICATION ==="
log_separator

log_info "Using rootfs path: $DISTRO_ROOT"
log_info "Verifying rootfs existence..."

if [ ! -d "$DISTRO_ROOT" ]; then
    log_error "Debian rootfs not found at: $DISTRO_ROOT"
    log_info "Contents of $DISTRO_MOUNT:"
    find "$DISTRO_MOUNT" -maxdepth 2 -type d 2>&1 | tee -a "$LOGFILE"
    umount "$DISTRO_MOUNT" 2>/dev/null
    sync
    exec /system/bin/realrecovery "$@"
fi

log_success "Debian rootfs directory exists"

log ""
log_info "Verifying critical rootfs directories..."
CRITICAL_DIRS="etc usr var home root boot dev proc sys tmp run"
MISSING_DIRS=""

for dir in $CRITICAL_DIRS; do
    if [ -d "$DISTRO_ROOT/$dir" ]; then
        log_success "Found: /$dir"
    else
        log_warning "Missing: /$dir"
        MISSING_DIRS="$MISSING_DIRS $dir"
    fi
done

if [ -n "$MISSING_DIRS" ]; then
    log_warning "Some directories are missing:$MISSING_DIRS"
    log_info "This may indicate an incomplete installation"
else
    log_success "All critical directories present"
fi

log ""
log_info "Checking Debian binary paths..."
log_subsection
log_info "Debian uses /usr/bin instead of /bin (usrmerge)"
if [ -d "$DISTRO_ROOT/bin" ]; then
    BIN_COUNT=$(find "$DISTRO_ROOT/bin" -maxdepth 1 -type f 2>/dev/null | wc -l)
    log_info "/bin exists with $BIN_COUNT+ entries"
fi

if [ -d "$DISTRO_ROOT/usr/bin" ]; then
    USRBIN_COUNT=$(find "$DISTRO_ROOT/usr/bin" -maxdepth 1 -type f 2>/dev/null | head -100 | wc -l)
    if [ "$USRBIN_COUNT" -eq 100 ]; then
        log_success "/usr/bin has 100+ binaries"
    else
        log_success "/usr/bin has $USRBIN_COUNT binaries"
    fi
else
    log_error "/usr/bin not found - critical issue"
fi

# Check for bash (su will be used with proper environment)
if [ -f "$DISTRO_ROOT/usr/bin/bash" ]; then
    log_success "bash found at /usr/bin/bash"
    BASH_SIZE=$(stat -c%s "$DISTRO_ROOT/usr/bin/bash" 2>/dev/null)
    log_info "bash binary size: $BASH_SIZE bytes"
    SHELL_CMD="/usr/bin/bash"
elif [ -f "$DISTRO_ROOT/bin/bash" ]; then
    log_success "bash found at /bin/bash"
    SHELL_CMD="/bin/bash"
else
    log_warning "bash not found, using sh"
    SHELL_CMD="/bin/sh"
fi
log_info "Will use shell: $SHELL_CMD"
log_subsection

log ""
log_info "Checking user configuration..."
log_info "Using user '$DEBIAN_USER' from /home/$DEBIAN_USER"
if [ -d "$DISTRO_ROOT/home/$DEBIAN_USER" ]; then
    log_success "User home directory exists: /home/$DEBIAN_USER"
else
    log_warning "User home directory not found: /home/$DEBIAN_USER"
fi

if grep -q "^$DEBIAN_USER:" "$DISTRO_ROOT/etc/passwd" 2>/dev/null; then
    USER_INFO=$(grep "^$DEBIAN_USER:" "$DISTRO_ROOT/etc/passwd")
    log_success "User exists in /etc/passwd"
    log_info "User entry: $USER_INFO"
else
    log_warning "User '$DEBIAN_USER' not found in /etc/passwd"
fi

log ""
log_info "Rootfs statistics:"
DIR_COUNT=$(find "$DISTRO_ROOT" -type d 2>/dev/null | wc -l)
FILE_COUNT=$(find "$DISTRO_ROOT" -type f 2>/dev/null | wc -l)
log_info "Total directories: $DIR_COUNT"
log_info "Total files: $FILE_COUNT"
log ""

# === MESA TARBALL CHECK AND TRANSFER ===
log_separator
log "=== PHASE 5B: MESA INSTALLATION PREPARATION ==="
log_separator
log_info "Version 5.0 Addition: Mesa 25.3.0 Android container installation"
log ""

log_info "Checking for Mesa tarball: $MESA_TARBALL"
MESA_SD_PATH="$SD_MOUNT/$MESA_TARBALL"
MESA_DISTRO_PATH="$DISTRO_ROOT/$MESA_TARBALL"

if [ -f "$MESA_DISTRO_PATH" ]; then
    MESA_SIZE=$(stat -c%s "$MESA_DISTRO_PATH" 2>/dev/null || echo "0")
    MESA_MB=$((MESA_SIZE / 1024 / 1024))
    log_success "Mesa tarball already in Debian root: ${MESA_MB}MB"
    log_info "Location: $MESA_DISTRO_PATH"
elif [ -f "$MESA_SD_PATH" ]; then
    MESA_SIZE=$(stat -c%s "$MESA_SD_PATH" 2>/dev/null || echo "0")
    MESA_MB=$((MESA_SIZE / 1024 / 1024))
    log_info "Mesa tarball found on SD card: ${MESA_MB}MB"
    log_info "Source: $MESA_SD_PATH"
    log_info "Destination: $MESA_DISTRO_PATH"
    
    log_attempt "Copying Mesa tarball to Debian root..."
    COPY_START=$(date +%s)
    
    if cp "$MESA_SD_PATH" "$MESA_DISTRO_PATH" 2>&1 | tee -a "$LOGFILE"; then
        COPY_END=$(date +%s)
        COPY_TIME=$((COPY_END - COPY_START))
        log_success "Mesa tarball copied successfully"
        log_info "Copy time: ${COPY_TIME} seconds"
        
        # Verify copy
        if [ -f "$MESA_DISTRO_PATH" ]; then
            COPIED_SIZE=$(stat -c%s "$MESA_DISTRO_PATH" 2>/dev/null || echo "0")
            COPIED_MB=$((COPIED_SIZE / 1024 / 1024))
            log_success "Verified: ${COPIED_MB}MB copied"
        else
            log_error "Copy verification failed"
        fi
    else
        log_error "Failed to copy Mesa tarball"
        log_warning "Mesa installation will be skipped"
    fi
else
    log_warning "Mesa tarball not found on SD card"
    log_info "Expected location: $MESA_SD_PATH"
    log_info "Mesa installation will be skipped"
    log_info "To install Mesa, place tarball on SD card root"
fi
log ""

# === TEMPORARY DIRECTORY HANDLING ===
log_info "Version 4.0+ Fix: Checking for existing temporary directory"
TEMP_DIR="$DISTRO_MOUNT/super_temp"

if [ -d "$TEMP_DIR" ]; then
    TEMP_SIZE=$(du -sk "$TEMP_DIR" 2>/dev/null | awk '{print $1}')
    log_warning "Temporary directory already exists: $TEMP_DIR"
    log_info "Existing size: ${TEMP_SIZE}KB"
    log_info "Reusing existing temporary directory"
    
    # Check for existing vendor image
    if [ -f "$TEMP_DIR/vendor.img" ]; then
        VENDOR_IMG="$TEMP_DIR/vendor.img"
        IMG_SIZE=$(stat -c%s "$VENDOR_IMG" 2>/dev/null || echo "0")
        IMG_MB=$((IMG_SIZE / 1024 / 1024))
        log_warning "Existing vendor image found: ${IMG_MB}MB"
        log_info "Will attempt to reuse existing vendor image"
    fi
else
    log_info "Creating new temporary directory: $TEMP_DIR"
    mkdir -p "$TEMP_DIR" 2>/dev/null
    
    if [ -d "$TEMP_DIR" ]; then
        log_success "Temporary directory created"
    else
        log_error "Failed to create temporary directory"
    fi
fi
log ""

# === MOUNT PSEUDO FILESYSTEMS ===
log_separator
log "=== PHASE 6: PSEUDO FILESYSTEM MOUNTING ==="
log_separator

log_info "Version 4.0+ Fix: Checking virtual filesystem existence before creation"
log_info "Mounting pseudo filesystems for chroot environment..."
log ""

# /proc
log_attempt "Mounting /proc filesystem..."
mkdir -p "$DISTRO_ROOT/proc" 2>/dev/null
if mount | grep -q "$DISTRO_ROOT/proc"; then
    log_warning "/proc already mounted, skipping"
elif mount -t proc proc "$DISTRO_ROOT/proc" 2>&1 | tee -a "$LOGFILE"; then
    log_success "/proc mounted successfully"
    log_info "Mount verification: $(mount | grep 'proc on' | grep "$DISTRO_ROOT/proc")"
else
    log_error "Failed to mount /proc"
fi
log ""

# /sys
log_attempt "Mounting /sys filesystem..."
mkdir -p "$DISTRO_ROOT/sys" 2>/dev/null
if mount | grep -q "$DISTRO_ROOT/sys"; then
    log_warning "/sys already mounted, skipping"
elif mount -t sysfs sysfs "$DISTRO_ROOT/sys" 2>&1 | tee -a "$LOGFILE"; then
    log_success "/sys mounted successfully"
    log_info "Mount verification: $(mount | grep 'sysfs on' | grep "$DISTRO_ROOT/sys")"
else
    log_error "Failed to mount /sys"
fi
log ""

# /dev
log_attempt "Bind mounting /dev filesystem..."
mkdir -p "$DISTRO_ROOT/dev" 2>/dev/null
if mount | grep -q "$DISTRO_ROOT/dev"; then
    log_warning "/dev already mounted, skipping"
elif mount --bind /dev "$DISTRO_ROOT/dev" 2>&1 | tee -a "$LOGFILE"; then
    log_success "/dev bind mounted successfully"
    DEV_COUNT=$(ls "$DISTRO_ROOT/dev" 2>/dev/null | wc -l)
    log_info "Device nodes available: $DEV_COUNT entries"
else
    log_error "Failed to bind mount /dev"
fi
log ""

# /dev/pts
log_attempt "Mounting /dev/pts filesystem..."
mkdir -p "$DISTRO_ROOT/dev/pts" 2>/dev/null
if mount | grep -q "$DISTRO_ROOT/dev/pts"; then
    log_warning "/dev/pts already mounted, skipping"
elif mount -t devpts devpts "$DISTRO_ROOT/dev/pts" -o noexec,nosuid,gid=5,mode=620 2>&1 | tee -a "$LOGFILE"; then
    log_success "/dev/pts mounted successfully"
    log_info "Mount options: noexec,nosuid,gid=5,mode=620"
else
    log_error "Failed to mount /dev/pts"
fi
log ""

# /dev/shm
log_attempt "Mounting /dev/shm filesystem..."
log_info "Version 4.0+ Fix: Checking if /dev/shm exists in chroot"
if [ ! -d "$DISTRO_ROOT/dev/shm" ]; then
    log_info "Creating /dev/shm directory"
    mkdir -p "$DISTRO_ROOT/dev/shm" 2>/dev/null
fi

if mount | grep -q "$DISTRO_ROOT/dev/shm"; then
    log_warning "/dev/shm already mounted, skipping"
elif mount -t tmpfs tmpfs "$DISTRO_ROOT/dev/shm" -o nosuid,nodev 2>&1 | tee -a "$LOGFILE"; then
    log_success "/dev/shm mounted successfully"
    log_info "Mount options: nosuid,nodev"
else
    log_warning "Failed to mount /dev/shm (optional, may not be critical)"
fi
log ""

# /tmp
log_attempt "Mounting /tmp filesystem..."
mkdir -p "$DISTRO_ROOT/tmp" 2>/dev/null
if mount | grep -q "$DISTRO_ROOT/tmp"; then
    log_warning "/tmp already mounted, skipping"
elif mount -t tmpfs tmpfs "$DISTRO_ROOT/tmp" -o nosuid,nodev 2>&1 | tee -a "$LOGFILE"; then
    log_success "/tmp mounted as tmpfs"
    log_info "Mount options: nosuid,nodev"
else
    log_warning "/tmp mount failed, will use existing directory"
fi
log ""

# /run
log_attempt "Mounting /run filesystem..."
log_info "Version 4.0+ Fix: /run is a tmpfs in RAM, creating if needed"
if [ ! -d "$DISTRO_ROOT/run" ]; then
    log_info "Creating /run directory"
    mkdir -p "$DISTRO_ROOT/run" 2>/dev/null
fi

if mount | grep -q "$DISTRO_ROOT/run"; then
    log_warning "/run already mounted, skipping"
elif [ -d "/run" ] && mount --bind /run "$DISTRO_ROOT/run" 2>&1 | tee -a "$LOGFILE"; then
    log_success "/run bind mounted from host"
else
    log_info "Host /run unavailable, creating tmpfs"
    if mount -t tmpfs tmpfs "$DISTRO_ROOT/run" -o nosuid,nodev,mode=0755 2>&1 | tee -a "$LOGFILE"; then
        log_success "/run mounted as tmpfs"
    else
        log_warning "/run mount failed (optional)"
    fi
fi
log ""

# /run/shm (if needed by Debian)
log_attempt "Checking /run/shm (Debian runtime requirement)..."
if [ ! -d "$DISTRO_ROOT/run/shm" ]; then
    log_info "Creating /run/shm directory"
    mkdir -p "$DISTRO_ROOT/run/shm" 2>/dev/null
fi

if mount | grep -q "$DISTRO_ROOT/run/shm"; then
    log_warning "/run/shm already mounted, skipping"
elif mount -t tmpfs tmpfs "$DISTRO_ROOT/run/shm" -o nosuid,nodev 2>&1 | tee -a "$LOGFILE"; then
    log_success "/run/shm mounted as tmpfs"
else
    log_info "/run/shm mount failed (may not be required)"
fi
log ""

log_info "Pseudo filesystem mounting summary:"
MOUNTED_COUNT=$(mount | grep "$DISTRO_ROOT" | wc -l)
log_info "Total filesystems mounted in chroot: $MOUNTED_COUNT"
log ""

# === EXTRACT AND MOUNT VENDOR FROM SUPER PARTITION ===
log_separator
log "=== PHASE 7: VENDOR PARTITION EXTRACTION ==="
log_separator

VENDOR_IMG="${TEMP_DIR}/vendor.img"

if [ -f "$VENDOR_IMG" ]; then
    IMG_SIZE=$(stat -c%s "$VENDOR_IMG" 2>/dev/null || echo "0")
    IMG_MB=$((IMG_SIZE / 1024 / 1024))
    log_info "Using existing vendor image: ${IMG_MB}MB"
    log_info "Skipping extraction to save time"
else
    log_info "Extracting vendor partition from super partition..."
    log_info "Super partition device: $SUPER_PARTITION"

    # Vendor partition location in super (sector addresses)
    VENDOR_START=3436544
    VENDOR_SIZE=3700976

    log_info "Vendor partition specifications:"
    log_info "  Start sector: $VENDOR_START"
    log_info "  Size (sectors): $VENDOR_SIZE"
    log_info "  Block size: 4096 bytes"
    log_info "  Skip blocks: $((VENDOR_START / 8))"
    log_info "  Count blocks: $((VENDOR_SIZE / 8))"

    log_info "Target image file: $VENDOR_IMG"
    log ""

    log_attempt "Starting dd extraction (this may take a moment)..."
    DD_START=$(date +%s)

    if dd if="$SUPER_PARTITION" of="$VENDOR_IMG" bs=4096 skip=$((VENDOR_START / 8)) count=$((VENDOR_SIZE / 8)) 2>&1 | tee -a "$LOGFILE"; then
        DD_END=$(date +%s)
        DD_TIME=$((DD_END - DD_START))
        
        log_success "Vendor image extracted successfully"
        log_info "Extraction time: ${DD_TIME} seconds"
        
        if [ -f "$VENDOR_IMG" ]; then
            IMG_BYTES=$(stat -c%s "$VENDOR_IMG" 2>/dev/null || echo "0")
            IMG_MB=$((IMG_BYTES / 1024 / 1024))
            log_info "Image file size: ${IMG_MB}MB ($IMG_BYTES bytes)"
            log_info "Image file location: $VENDOR_IMG"
        else
            log_error "Image file not found after extraction"
        fi
    else
        log_error "Failed to extract vendor image"
        log_warning "Vendor partition will not be available"
    fi
fi
log ""

# Mount vendor image via loop device
if [ -f "$VENDOR_IMG" ]; then
    log_attempt "Mounting vendor image via loop device..."
    
    VENDOR_LOOP=$(losetup -f 2>/dev/null)
    if [ -z "$VENDOR_LOOP" ]; then
        log_error "No free loop device available"
        log_info "Current loop devices:"

    rm -f "$TEMP_DIR/vendor.img" 2>&1 | tee -a "$LOGFILE" && log_success "Vendor image removed"
    
    log_attempt "Removing temporary directory..."
    rm -rf "$TEMP_DIR" 2>&1 | tee -a "$LOGFILE" && log_success "Temporary directory removed" || log_warning "Failed to remove temporary directory"
    
    # Verify removal
    if [ ! -d "$TEMP_DIR" ]; then
        log_success "Temporary directory completely removed"
    else
        log_warning "Temporary directory still exists, forcing removal..."
        rm -rf "$TEMP_DIR"
    fi
else
    log_info "No temporary directory to clean"
fi

log ""
log_attempt "Unmounting SOCKO partition..."
umount "$SOCKO_MOUNT" 2>&1 | tee -a "$LOGFILE" && log_success "SOCKO unmounted" || log_info "SOCKO already unmounted"

log ""
log_attempt "Unmounting distro partition..."
umount "$DISTRO_MOUNT" 2>&1 | tee -a "$LOGFILE" && log_success "Distro partition unmounted" || log_warning "Distro partition unmount had issues"

log ""
log_info "Syncing filesystems to ensure all data is written..."
sync
log_success "Filesystem sync complete"

log ""
log_info "Verifying cleanup..."
REMAINING_MOUNTS=$(mount | grep -c "$DISTRO_ROOT" 2>/dev/null || echo "0")
log_info "Remaining mounts in chroot: $REMAINING_MOUNTS"

if [ "$REMAINING_MOUNTS" -eq 0 ]; then
    log_success "All chroot mounts cleaned up successfully"
else
    log_warning "Some mounts may still be active:"
    mount | grep "$DISTRO_ROOT" | tee -a "$LOGFILE"
fi

log ""

# === SESSION COMPLETE ===
log_separator
log "=== SESSION COMPLETE ==="
log_separator
log ""
log_success "All phases completed successfully"
log ""
log_info "Generated Files:"
log_info "  Main log:         $LOGFILE"
log_info "  Hardware report:  $TEST_OUTPUT"
log_info "  Package list:     $PACKAGE_LIST"
log ""

if [ -f "$LOGFILE" ]; then
    LOG_SIZE_KB=$(stat -c%s "$LOGFILE" 2>/dev/null | awk '{print int($1/1024)}' || echo "0")
    log_info "Main log size: ${LOG_SIZE_KB}KB"
fi

if [ -f "$TEST_OUTPUT" ]; then
    TEST_SIZE_KB=$(stat -c%s "$TEST_OUTPUT" 2>/dev/null | awk '{print int($1/1024)}' || echo "0")
    log_info "Hardware report size: ${TEST_SIZE_KB}KB"
fi

if [ -f "$PACKAGE_LIST" ]; then
    PKG_SIZE_KB=$(stat -c%s "$PACKAGE_LIST" 2>/dev/null | awk '{print int($1/1024)}' || echo "0")
    log_info "Package list size: ${PKG_SIZE_KB}KB"
fi

log ""
log_info "All logs are located in: $SD_MOUNT/logs/"
log ""
log_separator
log_info "Exiting to recovery in 5 seconds..."
log_separator
log ""

sleep 5

exec /system/bin/realrecovery "$@"
