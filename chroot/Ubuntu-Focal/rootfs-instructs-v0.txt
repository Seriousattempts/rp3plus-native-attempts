# Step 17
sudo chroot mnt /usr/bin/qemu-aarch64-static /bin/bash <<'EOF'
set -e

# ── 1. The Vulkan 1.3 Header Bypass ──
# Focal's libvulkan-dev is too old. We need modern headers to compile this.
rm -rf /tmp/Vulkan-Headers
git clone --depth 1 --branch main https://github.com/KhronosGroup/Vulkan-Headers.git /tmp/Vulkan-Headers

# Overwrite the system's ancient headers with the modern ones
cp -r /tmp/Vulkan-Headers/include/vulkan/* /usr/include/vulkan/
mkdir -p /usr/include/vk_video
cp -r /tmp/Vulkan-Headers/include/vk_video/* /usr/include/vk_video/ 2>/dev/null || true
rm -rf /tmp/Vulkan-Headers

# ── 2. Prepare lsfg-vk ──
rm -rf /tmp/lsfg-vk
git init /tmp/lsfg-vk
cd /tmp/lsfg-vk
git remote add origin https://github.com/PancakeTAS/lsfg-vk
git fetch --depth 1 origin ff1a0f72a7d6d08b84d58b7b4dc5f05c9f904f98
git checkout FETCH_HEAD
git submodule update --init --recursive

sed -i '1s|^|#include <optional>\n|' include/config/config.hpp

# Proactively strip the GUI tool from CMake
sed -i '/add_subdirectory(.*ui.*)/Id' CMakeLists.txt 2>/dev/null || true
sed -i '/find_package(.*GTK4.*)/Id' CMakeLists.txt 2>/dev/null || true

# ── 3. Compile the core Vulkan layer headless ──
cmake -B build -G Ninja \
  -DCMAKE_BUILD_TYPE=Release \
  -DCMAKE_C_COMPILER=clang-16 \
  -DCMAKE_CXX_COMPILER=clang++-16 \
  -DCMAKE_CXX_FLAGS="-stdlib=libc++ -fexperimental-library" \
  -DCMAKE_EXE_LINKER_FLAGS="-stdlib=libc++ -lc++abi" \
  -DCMAKE_SHARED_LINKER_FLAGS="-stdlib=libc++ -lc++abi" \
  -DCMAKE_INTERPROCEDURAL_OPTIMIZATION=OFF \
  -DBUILD_UI=OFF \
  -DENABLE_GUI=OFF

cmake --build build
cmake --install build --prefix /usr/local

cd /
rm -rf /tmp/lsfg-vk
EOF

# ── 1. Extract Standard AppImages via Offset Bypass ──
for name in xemu duckstation ryubing eden rpcs3; do
    cd /tmp
    echo "Extracting ${name}..."
    # Find the exact byte offset where the SquashFS payload begins
    OFFSET=$(grep -b -a -m 1 -o 'hsqs' "${name}.AppImage" | cut -d: -f1)
    
    if [ -n "$OFFSET" ]; then
        unsquashfs -f -o $OFFSET -d /opt/emulators/${name} "${name}.AppImage"
        ln -sf /opt/emulators/${name}/AppRun /usr/local/bin/${name}
    else
        echo "Error: Could not find SquashFS header in ${name}.AppImage"
    fi
    rm -f ${name}.AppImage
done

OFFSET=$(grep -b -a -m 1 -o 'hsqs' "$MELONDS_BIN" | cut -d: -f1)

    unsquashfs -f -o $OFFSET -d /opt/emulators/melonds "$MELONDS_BIN"
    ln -sf /opt/emulators/melonds/AppRun /usr/local/bin/melonds

#!/bin/sh
export WINEPREFIX="\$HOME/.wine-desmume"
export WINEARCH=win64
export BOX64_NOBANNER=1
if [ ! -d "\$WINEPREFIX" ]; then
    box64 wineboot --init
    box64 wine "$PREREQS_EXE" /SILENT 2>/dev/null || \
    WINEPREFIX="\$HOME/.wine-desmume" box64 winetricks -q vcrun2022
fi
exec box64 wine64 "$DESMUME_EXE" "\$@"
SCRIPT
chmod +x /usr/local/bin/desmume-win

#!/bin/bash

flatpak remote-add --system --if-not-exists dolphin https://flatpak.dolphin-emu.org/releases.flatpakrepo

flatpak install --system --noninteractive flathub org.azahar_emu.Azahar
flatpak install --system --noninteractive dolphin org.DolphinEmu.dolphin-emu

SCRIPT
