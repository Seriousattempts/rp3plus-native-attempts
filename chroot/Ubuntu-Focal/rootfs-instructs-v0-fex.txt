sudo chroot mnt /usr/bin/qemu-aarch64-static /bin/bash <<'EOF'
export DEBIAN_FRONTEND=noninteractive

# Protect QEMU from running out of memory
export MAKEFLAGS="-j2"

# ── 1. Install Modern CMake (Kitware PPA) & Dependencies ──────────────
apt-get update
apt-get install -y apt-transport-https ca-certificates gnupg software-properties-common wget

    build-essential git unzip \
    libsfml-dev libopenal-dev libflac-dev \
    libvorbis-dev libjpeg-dev libfreetype6-dev \
    libudev-dev libfontconfig1-dev libpng-dev \
    xwayland cmake

# ── 2. Download and Extract Christian Haitian's ARM64 Cores ───────────
echo "Fetching Christian Haitian's aarch64 Libretro cores..."
rm -rf /opt/cores_temp
mkdir -p /opt/cores_temp

# Standard shallow clone (Compatible with Git 2.25)
git clone --depth 1 https://github.com/christianhaitian/retroarch-cores.git /opt/cores_temp/retroarch-cores

# Create the RetroArch core directory
mkdir -p /root/.config/retroarch/cores
mkdir -p /root/.config/retroarch/cores/info

# Unzip all the cores into the RetroArch directory
echo "Extracting cores..."
for zip_file in /opt/cores_temp/retroarch-cores/aarch64/*.zip; do
    unzip -q -o "$zip_file" -d /root/.config/retroarch/cores/
done

# Cleanup the massive git repo
cd /
rm -rf /opt/cores_temp
echo "Libretro cores successfully staged."

# ── 3. Build AttractMode Plus (v3.2.3) ────────────────────────────────
echo "Building AttractMode Plus..."

rm -rf attractplus
# Clone exactly tag 3.2.3
git clone --depth 1 --branch 3.2.3 https://github.com/oomek/attractplus.git
cd attractplus

# Compile using GLES (OpenGL ES) to match the Mali GPU, bypassing GBM/DRM
make USE_GLES=1 -j2
make install

# ── 4. Create the Mir -> AttractMode Launcher ─────────────────────────
cat > /usr/local/bin/mir-gpu <<'LAUNCH'
#!/bin/bash
export XDG_RUNTIME_DIR="/run/user/$(id -u)"
mkdir -p "$XDG_RUNTIME_DIR"
chmod 0700 "$XDG_RUNTIME_DIR"

mkdir -p /run/dbus

    dbus-daemon --system --fork

# Load Android GPU drivers
source /etc/profile.d/mir-hybris.sh 2>/dev/null || true
export WAYLAND_DISPLAY=wayland-0

echo "Launching AttractMode Plus via Mir Compositor..."
# Mir will automatically spin up Xwayland for AttractMode
exec miral-kiosk "attract" "$@"
LAUNCH
chmod 755 /usr/local/bin/mir-gpu

# ── 5. Cleanup ────────────────────────────────────────────────────────
apt-get clean
rm -rf /var/cache/apt/archives/*

echo "════ AttractMode Plus & Libretro Cores Installed ════"
EOF

export DEBIAN_FRONTEND=noninteractive

apt-get update
apt-get install -y proot wget tar

mkdir -p /opt/modern-os
cd /opt/modern-os

# 1. Download an official Ubuntu 24.04 ARM64 Base Rootfs (Contains modern GLIBC and SDL2)
echo "Downloading modern Ubuntu 24.04 sandbox..."
wget -qO noble-base.tar.gz http://cdimage.ubuntu.com/ubuntu-base/releases/24.04/release/ubuntu-base-24.04-base-arm64.tar.gz

# 2. Extract it
tar -xf noble-base.tar.gz
rm noble-base.tar.gz

# 3. Copy DNS resolution so the container has internet
cp /etc/resolv.conf etc/resolv.conf

# 4. Install Pipewire, modern SDL2, and Wayland clients INSIDE the sandbox
echo "Installing modern dependencies inside the sandbox..."
proot -R /opt/modern-os -0 -b /dev -b /sys -b /proc /bin/bash -c "
  apt-get update && \
  apt-get install -y libpipewire-0.3-0 libsdl2-2.0-0 libwayland-client0 libvulkan1 \
                     libx11-6 libxrandr2 libxcb1 libxcursor1 libpulse0 fuse
"

# 5. Create a Universal Proot Wrapper Script
# This script tricks any AppImage into thinking it's running on Ubuntu 24.04
cat > /usr/local/bin/run-modern <<'WRAPPER'
#!/bin/bash
APP_PATH="$1"
shift

# Bind mount the host's Wayland socket, GPU nodes, and Home directory into the container,
# then execute the app using the container's modern GLIBC.
exec proot -R /opt/modern-os \
  -b /dev:/dev \
  -b /sys:/sys \
  -b /proc:/proc \
  -b /tmp:/tmp \
  -b /run/user:/run/user \
  -b "$HOME:$HOME" \
  -w "$HOME" \
  "$APP_PATH" "$@"
WRAPPER
chmod +x /usr/local/bin/run-modern

echo "════ Modern OS Sandbox Created ════"
EOF

  /usr/lib/aarch64-linux-gnu/libandroid-properties.so* \

# ── 3. Register GLVND EGL ICD for libhybris ───────────────────────────

ln -sf /opt/libhybris/share/glvnd/egl_vendor.d/10_libhybris.json \
       /usr/share/glvnd/egl_vendor.d/10_libhybris.json

# ── 4. Add UBports Repository ─────────────────────────────────────────
# Required for libmirplatform18 instead of Ubuntu's older v17
echo "deb [trusted=yes] http://repo.ubports.com focal main" \
  > /etc/apt/sources.list.d/ubports.list
apt-get update -qq

# ── 5. Install Base Mir Dependencies ──────────────────────────────────
apt-get install -y \
  libmirclient-dev \
  libmiral-dev \
  libmirplatform18 \
  libmirplatform-dev \
  libmircommon7 \
  mir-platform-input-evdev7 \
  mir-platform-graphics-mesa-kms16 \
  mir-platform-graphics-mesa-x16 \
  mir-demos

# ── 6. Install Custom Android Graphics Modules ────────────────────────

dpkg -i /tmp/mir1-client-platform-android5_0.31.2_arm64.deb \
        /tmp/mir1-platform-graphics-android16_0.31.2_arm64.deb \
  || apt-get install -f -y
rm -f /tmp/mir1-*.deb

# ── 7. Symlink /lib entries back to /opt/libhybris ────────────────────
# apt-get -f above reinstalls UBports libhybris debs into /lib/aarch64-linux-gnu/.
# Override them with symlinks so the custom /opt/libhybris build always wins.
for lib in \
  libhybris-common.so.1.0.0 \
  libhybris-common.so.1 \
  libhardware.so.2.0.0 \
  libhardware.so.2 \
  libandroid-properties.so.1.0.0 \
  libandroid-properties.so.1 \
  libEGL_libhybris.so.0.0.0 \
  libEGL_libhybris.so.0; do
    [ -f "/opt/libhybris/lib/$lib" ] && \
      ln -sf "/opt/libhybris/lib/$lib" "/lib/aarch64-linux-gnu/$lib" && \
      echo "Linked $lib"
done
ldconfig

# ── 8. Set Up Hardware Access Groups ──────────────────────────────────
groupadd -f seat
groupadd -f fuse
for U in user phablet; do
  id "$U" &>/dev/null && usermod -aG video,render,input,seat,tty,fuse,audio "$U" || true
done

rm -rf /var/cache/apt/archives/*

echo "════ Mir + libhybris Stack Successfully Installed & Configured ════"
EOF
