#!/system/bin/sh

# ---- Logging setup ----
BOOT_LOG="/boot/void_boot.log"
FALLBACK_LOG="/void_boot.log"

# Try to use boot partition for logging
if touch "$BOOT_LOG".test 2>/dev/null; then
    LOGFILE="$BOOT_LOG"
    rm -f "$BOOT_LOG".test
else
    LOGFILE="$FALLBACK_LOG"
fi

# Clear existing log
> "$LOGFILE"

# Logging function
log() {
    TS="$(/system/bin/uptime 2>/dev/null | cut -d' ' -f1 2>/dev/null)"
    [ -z "$TS" ] && TS="T"
    echo "[$TS] $*" | tee -a "$LOGFILE" >/dev/null
}

log "=== Void Linux aarch64 Boot Script Started ==="

# ---- Start ueventd for device filesystem ----
log "INIT: Starting ueventd for device node creation"
if [ -f /system/bin/ueventd ]; then
    /system/bin/ueventd &
    sleep 2
    log "INIT: ueventd started successfully"
else
    log "ERROR: ueventd not found"
    exec /system/bin/realrecovery "$@"
fi

# ---- Setup binderfs ----
log "INIT: Setting up binderfs"
if [ -d /dev/binderfs ]; then
    ln -sf /dev/binderfs/binder /dev/binder 2>/dev/null
    ln -sf /dev/binderfs/hwbinder /dev/hwbinder 2>/dev/null
    ln -sf /dev/binderfs/vndbinder /dev/vndbinder 2>/dev/null
    log "INIT: binderfs symlinks created"
else
    if mkdir -p /dev/binderfs 2>/dev/null; then
        mount -t binder binder /dev/binderfs 2>/dev/null
        chmod 755 /dev/binderfs 2>/dev/null
        ln -sf /dev/binderfs/binder /dev/binder 2>/dev/null
        ln -sf /dev/binderfs/hwbinder /dev/hwbinder 2>/dev/null
        ln -sf /dev/binderfs/vndbinder /dev/vndbinder 2>/dev/null
        log "INIT: binderfs mounted and configured"
    else
        log "WARNING: Failed to setup binderfs"
    fi
fi

# ---- Boot partition configuration ----
BOOT_PARTITION_MOUNT_POINT=/boot
BOOT_PARTITION=/dev/block/mmcblk1p1
BOOT_PARTITION_WAIT_COUNTER=0
ROOTFS_IMG="/boot/ext3fs.img"
INIT_SCRIPT="/boot/init"

log "BOOT: Waiting for boot partition $BOOT_PARTITION"

# ---- Wait for boot partition detection ----
while [ ! -b $BOOT_PARTITION ]; do
    sleep 0.1
    BOOT_PARTITION_WAIT_COUNTER=$((BOOT_PARTITION_WAIT_COUNTER + 1))
    if [ $BOOT_PARTITION_WAIT_COUNTER -ge 50 ]; then
        log "ERROR: Boot partition not detected after 5 seconds"
        exec /system/bin/realrecovery "$@"
    fi
done

log "BOOT: Boot partition $BOOT_PARTITION detected"

# ---- Mount boot partition ----
mkdir -p $BOOT_PARTITION_MOUNT_POINT
if mount -t exfat $BOOT_PARTITION $BOOT_PARTITION_MOUNT_POINT; then
    log "BOOT: Boot partition mounted at $BOOT_PARTITION_MOUNT_POINT"
    
    # Ensure boot is writable
    mount -o remount,rw $BOOT_PARTITION_MOUNT_POINT 2>/dev/null
    log "BOOT: Boot partition remounted as read-write"
    
    # Update log file location now that boot is mounted
    if [ "$LOGFILE" = "$FALLBACK_LOG" ]; then
        LOGFILE="$BOOT_LOG"
        cat "$FALLBACK_LOG" > "$LOGFILE" 2>/dev/null
        log "LOG: Switched to boot partition logging"
    fi
else
    log "ERROR: Failed to mount boot partition"
    exec /system/bin/realrecovery "$@"
fi

# ---- Check for required files ----
if [ ! -f "$BOOT_PARTITION_MOUNT_POINT/$ROOTFS_IMG" ]; then
    log "ERROR: Void Linux rootfs image not found: $BOOT_PARTITION_MOUNT_POINT/$ROOTFS_IMG"
    umount $BOOT_PARTITION_MOUNT_POINT
    exec /system/bin/realrecovery "$@"
fi

if [ ! -f "$BOOT_PARTITION_MOUNT_POINT/$INIT_SCRIPT" ]; then
    log "ERROR: Init script not found: $BOOT_PARTITION_MOUNT_POINT/$INIT_SCRIPT"
    umount $BOOT_PARTITION_MOUNT_POINT
    exec /system/bin/realrecovery "$@"
fi

log "VOID: Found Void Linux rootfs: $BOOT_PARTITION_MOUNT_POINT/$ROOTFS_IMG"
log "INIT: Found init script: $BOOT_PARTITION_MOUNT_POINT/$INIT_SCRIPT"

# ---- Mount ext3fs.img rootfs ----
mkdir -p /mnt/void
if mount -o loop "$BOOT_PARTITION_MOUNT_POINT/$ROOTFS_IMG" /mnt/void; then
    log "VOID: Rootfs mounted successfully at /mnt/void"
else
    log "ERROR: Failed to mount Void Linux rootfs"
    umount $BOOT_PARTITION_MOUNT_POINT
    exec /system/bin/realrecovery "$@"
fi

# ---- Set SELinux to permissive immediately ----
if [ -f /sys/fs/selinux/enforce ]; then
    echo 0 > /sys/fs/selinux/enforce 2>/dev/null
    log "SELINUX: Set to permissive mode"
else
    log "WARNING: SELinux enforce file not found"
fi

# ---- Enable adbd for debugging ----
log "ADB: Enabling adbd for debugging"
start adbd
setprop sys.usb.config adb
setprop sys.usb.state adb
setprop sys.usb.ffs.ready 1
log "ADB: adbd enabled"

# ---- Reset boot device to emmc (safety) ----
/system/bin/erasebootrecord 2>/dev/null
log "BOOT: Reset boot device to emmc"

# ---- Extract initramfs and set up proper environment ----
log "INITRAMFS: Extracting initramfs to set up proper init environment"
INITRAMFS_PATH="/mnt/void/boot/initramfs-6.12.11_1.img"
INITRAMFS_MOUNT="/mnt/initramfs"
mkdir -p "$INITRAMFS_MOUNT"

if [ -f "$INITRAMFS_PATH" ]; then
    cd "$INITRAMFS_MOUNT"
    if /system/bin/busybox zcat "$INITRAMFS_PATH" | /system/bin/busybox cpio -i 2>/dev/null; then
        log "INITRAMFS: Extracted successfully to $INITRAMFS_MOUNT"
        
        # Verify critical initramfs components
        if [ -f "$INITRAMFS_MOUNT/init" ]; then
            log "INITRAMFS: Found init at $INITRAMFS_MOUNT/init"
        fi
        
        if [ -f "$INITRAMFS_MOUNT/usr/lib/dracut-lib.sh" ]; then
            log "INITRAMFS: Found dracut-lib.sh at $INITRAMFS_MOUNT/usr/lib/dracut-lib.sh"
        else
            log "WARNING: dracut-lib.sh not found in expected location"
        fi
    else
        log "ERROR: Failed to extract initramfs"
        umount /mnt/void
        umount $BOOT_PARTITION_MOUNT_POINT
        exec /system/bin/realrecovery "$@"
    fi
    cd /
else
    log "ERROR: Initramfs not found at $INITRAMFS_PATH"
    umount /mnt/void
    umount $BOOT_PARTITION_MOUNT_POINT
    exec /system/bin/realrecovery "$@"
fi

# ---- Set up chroot environment for init ----
log "VOID: Setting up chroot environment with initramfs as root"

# Mount essential filesystems in the initramfs root
mount -t proc none "$INITRAMFS_MOUNT/proc" 2>/dev/null && log "MOUNT: proc mounted in initramfs"
mount -t sysfs none "$INITRAMFS_MOUNT/sys" 2>/dev/null && log "MOUNT: sysfs mounted in initramfs"
mount --rbind /dev "$INITRAMFS_MOUNT/dev" 2>/dev/null && log "MOUNT: dev mounted in initramfs"
mkdir -p "$INITRAMFS_MOUNT/run"
mount --rbind /run "$INITRAMFS_MOUNT/run" 2>/dev/null && log "MOUNT: run mounted in initramfs"

# Mount the Void rootfs inside the initramfs environment
mkdir -p "$INITRAMFS_MOUNT/sysroot"
mount -o bind /mnt/void "$INITRAMFS_MOUNT/sysroot" 2>/dev/null && log "MOUNT: Void rootfs bound to /sysroot in initramfs"

# Mount boot partition for access
mkdir -p "$INITRAMFS_MOUNT/boot"
mount -o bind $BOOT_PARTITION_MOUNT_POINT "$INITRAMFS_MOUNT/boot" 2>/dev/null && log "MOUNT: Boot partition bound in initramfs"

# Copy resolv.conf for network access
if [ -f /system/etc/resolv.conf ]; then
    cp /system/etc/resolv.conf "$INITRAMFS_MOUNT/etc/" 2>/dev/null
elif [ -f /etc/resolv.conf ]; then
    cp /etc/resolv.conf "$INITRAMFS_MOUNT/etc/" 2>/dev/null
else
    mkdir -p "$INITRAMFS_MOUNT/etc"
    echo "nameserver 8.8.8.8" > "$INITRAMFS_MOUNT/etc/resolv.conf"
    echo "nameserver 1.1.1.1" >> "$INITRAMFS_MOUNT/etc/resolv.conf"
fi
log "NETWORK: Created resolv.conf in initramfs"

# ---- Configure hardware-specific parameters ----
log "CONFIG: Setting up hardware environment for UMS512"

# Set up kernel command line environment that init expects
mkdir -p "$INITRAMFS_MOUNT/proc"
echo "earlycon=sprd_serial,0x70100000,115200n8 console=ttyS1,115200n8 console=tty1 loglevel=4 init=/init root=/dev/mmcblk1p1 lcd_name=lcd_FC11339_mipi_hd lcd_size=1336x752 swiotlb=1 rw rootwait" > "$INITRAMFS_MOUNT/proc/cmdline"
log "CONFIG: Created cmdline with UMS512 parameters"

# ---- Execute the init script from SD card ----
log "INIT: Preparing to execute init script from SD card"
log "INIT: This will run the actual Void Linux init process"

# Copy the init script to the initramfs root as /init
cp "$BOOT_PARTITION_MOUNT_POINT/$INIT_SCRIPT" "$INITRAMFS_MOUNT/init"
chmod +x "$INITRAMFS_MOUNT/init"
log "INIT: Copied init script to initramfs root as /init"

# Set up environment variables that the init script expects
export PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
export HOME=/root
export TERM=linux

# ---- Execute init in chroot environment ----
log "INIT: Starting Void Linux init process in proper environment"
log "INIT: This should start the full Void Linux system with XFCE"

# Change to initramfs root and execute init
if chroot "$INITRAMFS_MOUNT" /init 2>&1 | while read line; do 
    log "INIT_OUTPUT: $line"
done; then
    log "INIT: Init script completed successfully"
else
    RC=$?
    log "ERROR: Init script failed with exit code $RC"
    log "FALLBACK: Starting realrecovery as fallback"
    
    # Clean up mounts before fallback
    umount "$INITRAMFS_MOUNT/boot" 2>/dev/null
    umount "$INITRAMFS_MOUNT/sysroot" 2>/dev/null
    umount "$INITRAMFS_MOUNT/run" 2>/dev/null
    umount "$INITRAMFS_MOUNT/dev" 2>/dev/null
    umount "$INITRAMFS_MOUNT/sys" 2>/dev/null
    umount "$INITRAMFS_MOUNT/proc" 2>/dev/null
    umount /mnt/void 2>/dev/null
    umount $BOOT_PARTITION_MOUNT_POINT 2>/dev/null
    
    # Execute real recovery as fallback
    exec /system/bin/realrecovery "$@"
fi

log "INIT: Void Linux init has completed, system should be running"
log "INIT: Connect via ADB to interact with the Void Linux system"
log "INIT: System will continue running until manually stopped"

# Keep the script running to maintain the chroot environment
while true; do
    sleep 60
    if ! kill -0 $$ 2>/dev/null; then
        log "SYSTEM: Init process monitoring loop ended"
        break
    fi
done

# ---- Cleanup (only reached if system shuts down) ----
log "CLEANUP: Cleaning up mount points"
umount "$INITRAMFS_MOUNT/boot" 2>/dev/null
umount "$INITRAMFS_MOUNT/sysroot" 2>/dev/null
umount "$INITRAMFS_MOUNT/run" 2>/dev/null
umount "$INITRAMFS_MOUNT/dev" 2>/dev/null
umount "$INITRAMFS_MOUNT/sys" 2>/dev/null
umount "$INITRAMFS_MOUNT/proc" 2>/dev/null
umount /mnt/void 2>/dev/null
umount $BOOT_PARTITION_MOUNT_POINT 2>/dev/null

sync
log "=== Void Linux boot script completed ==="

# Execute real recovery as final fallback
exec /system/bin/realrecovery "$@"
