set -e

apk add --no-cache git python3

BUILD=/home/gamepad_src
rm -rf "$BUILD"
mkdir -p "$BUILD"

git clone \
    --depth=1 \
    --filter=blob:none \
    --sparse \
    https://github.com/GarlicOS/buildroot.git \
    "$BUILD"

cd "$BUILD"
git sparse-checkout set package/gamepad/src
mv package/gamepad/src/* .
rm -rf package .git

# ── Patch 1: device path ────────────────────────────────────
sed -i \
  's|#define RETROID_INPUT_GAMEPAD "/dev/ttyS0"|#define RETROID_INPUT_GAMEPAD "/dev/rscom"|' \
  retroid_input.h
echo "P1 OK: $(grep RETROID_INPUT_GAMEPAD retroid_input.h)"

# ── Patch 2: baud rate ──────────────────────────────────────
sed -i \
  's/cfsetispeed(&uart_config, 0x1002)/cfsetispeed(\&uart_config, B115200)/' \
  retroid_input.c
sed -i \
  's/cfsetospeed(&uart_config, 0x1002)/cfsetospeed(\&uart_config, B115200)/' \
  retroid_input.c
echo "P2 OK: $(grep B115200 retroid_input.c)"

# ── Patch 3a: enable pocket3plus (sed, no Python needed) ────
# Find the commented-out line and uncomment it, then remove the
# "Disabled for now" line that follows it
sed -i \
  's|^\t\t// merge_retroid_input_inputs(merged_gamepad);|\t\tmerge_retroid_input_inputs(merged_gamepad);|' \
  gamepad.c
sed -i \
  '/\/\/ Disabled for now until it is tested on RP3+/d' \
  gamepad.c
echo "P3a OK: $(grep -A1 'pocket3plus' gamepad.c | tail -2)"

# ── Patch 3b: DEVICE_CODENAME fallback (sed line insert) ────
# Insert a fallback line immediately after the getenv() call.
# Use sed's 'a' (append after match) — no newline counting needed.
sed -i \
  '/char \* code_name = getenv("DEVICE_CODENAME");/a\\tif (!code_name) code_name = "pocket3plus"; /* RP3+ fallback */' \
  gamepad.c
echo "P3b OK: $(grep 'pocket3plus.*fallback' gamepad.c)"

# ── Patch 3c: stub audio pipes (Python — only two replaces) ─
# These are large multi-line blocks; Python handles them cleanly.
# We only remove the assert checks that were causing false failures.
python3 - <<'PYEOF'
with open('gamepad.c', 'r') as f:
    src = f.read()

# Stub control pipe while-loop
old_ctrl = '''	// We haven't created the audio provider control pipe yet
	while (control_fd < 0)
	{
		// Create the named pipe
		mkfifo(AUDIO_CONTROL_PIPE, 0666);

		// Open the named pipe for writing
		control_fd = open(AUDIO_CONTROL_PIPE, O_WRONLY | O_NONBLOCK);
	}'''
new_ctrl = '\t/* STUB: audio control pipe skipped in Alpine */\n\tcontrol_fd = -1;'

# Stub response pipe while-loop
old_resp = '''	// We haven't created the audio provider response pipe yet
	while (response_fd < 0)
	{
		// Create the named pipe
		mkfifo(AUDIO_RESPONSE_PIPE, 0666);

		// Open the named pipe for reading
		response_fd = open(AUDIO_RESPONSE_PIPE, O_RDONLY | O_NONBLOCK);
	}'''
new_resp = '\t/* STUB: audio response pipe skipped in Alpine */\n\tresponse_fd = -1;'

if old_ctrl in src:
    src = src.replace(old_ctrl, new_ctrl)
    print("P3c-ctrl OK")
else:
    print("P3c-ctrl WARN: pattern not found, skipping (non-fatal)")

if old_resp in src:
    src = src.replace(old_resp, new_resp)
    print("P3c-resp OK")
else:
    print("P3c-resp WARN: pattern not found, skipping (non-fatal)")

with open('gamepad.c', 'w') as f:
    f.write(src)
PYEOF

# ── Compile ─────────────────────────────────────────────────
gcc -O2 -Wall \
    -static \
    -o /home/retroid_gamepad \
    gamepad.c \
    retroid_input.c \
    rg405.c \
    rg35xxplus.c \
    -lpthread

echo ""
echo "=== Build result ==="
ls -lh /home/retroid_gamepad
file /home/retroid_gamepad
sudo rm -rf gamepad_src
echo "=== Done ==="
EOF
