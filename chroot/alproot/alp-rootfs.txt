Step 4
MIRROR="https://dl-cdn.alpinelinux.org/alpine"

ARCH="aarch64"

APK_STATIC_PKG=$(curl -fsSL "${MIRROR}/edge/main/${ARCH}/" \
  | grep -o 'apk-tools-static-[0-9][^"]*\.apk' | head -1)

curl -fsSL "${MIRROR}/edge/main/${ARCH}/${APK_STATIC_PKG}" -o /tmp/apk-tools-static.apk

tar -xzf /tmp/apk-tools-static.apk -C /tmp sbin/apk.static

chmod +x /tmp/sbin/apk.static

sudo mkdir -p mnt/etc/apk/keys

tar -tf /tmp/apk-tools-static.apk | grep '\.pub' | while read k; do
  tar -xzf /tmp/apk-tools-static.apk "$k" -O 2>/dev/null \
    | sudo tee "mnt/etc/apk/keys/$(basename "$k")" > /dev/null || true
done

sudo /tmp/sbin/apk.static \
  --arch "${ARCH}" \
  -X "${MIRROR}/edge/main" \
  -X "${MIRROR}/edge/community" \
  --keys-dir mnt/etc/apk/keys \
  -U --allow-untrusted \
  --root mnt \
  --initdb \
  add alpine-base

sudo cp /usr/bin/qemu-aarch64-static mnt/usr/bin/

sudo tee mnt/etc/resolv.conf > /dev/null <<'EOF'
nameserver 1.1.1.1
nameserver 8.8.8.8
EOF

sudo mount -t proc   proc   mnt/proc

sudo mount -t sysfs  sysfs  mnt/sys

echo "============================================================"
echo "STEP 1 — Verifying 'virpipe' in Compiled Binaries"
echo "============================================================"
echo "Searching /usr/local/lib for virpipe strings..."

# We disable 'set -e' temporarily so grep doesn't halt the script if it finds no matches in a specific file
set +e

found=0
# Use find with -L to follow the symlinks Mesa created during install
for file in $(find -L /usr/local/lib -name "*.so*"); do
    if strings "$file" | grep -iq "virpipe"; then
        echo "✅ Found 'virpipe' reference in: $file"
        found=1
    fi
done

if [ $found -eq 0 ]; then
    echo "⚠️ Could not find 'virpipe' strings in the binaries."
fi

echo "STEP 2 — Cleaning Up Build Directory"
echo "============================================================"

echo "✅ Build folder erased."

echo "============================================================"
echo "DONE! Verification and cleanup complete."
echo "============================================================"
EOF
