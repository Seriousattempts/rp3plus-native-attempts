launch_miral_app_test() {
    logsection "PHASE : WESTON DRM COMPOSITOR + XWAYLAND TEST"

    logi "Hardware snapshot before launch:"
    hardware_status
    wake_display_pipeline

    chmod 666 /dev/dri/card0       2>/dev/null || true
    chmod 666 /dev/mali0           2>/dev/null || true
    chmod 666 /dev/ion             2>/dev/null || true
    chmod -R 666 /dev/input        2>/dev/null || true

    mkdir -p "$STRACELOGDIR"             2>/dev/null || true
    mkdir -p "$ALPINE/tmp/strace_logs"   2>/dev/null || true

    if ! grep -q " $ALPINE/tmp/strace_logs " /proc/mounts 2>/dev/null; then
        mount --bind "$STRACELOGDIR" "$ALPINE/tmp/strace_logs" 2>/dev/null \
            && logok "strace log dir bind-mounted at /tmp/strace_logs in chroot" \
            || logw  "could not bind strace log dir into chroot"
    fi

    mkdir -p "$ALPINE/tmp/.X11-unix" 2>/dev/null || true
    chmod 1777 "$ALPINE/tmp/.X11-unix"  2>/dev/null || true
    logok "pre-created /tmp/.X11-unix in chroot (sticky 1777)"

    # ---------------------------------------------------------------
    # weston.ini — pixel-format=argb8888 forces the DPU atomic path
    # to use DRM_FORMAT_ARGB8888 (4 bpp with alpha) instead of
    # DRM_FORMAT_XRGB8888.  The Sprd DPU vendor kernel's
    # drm_format_plane_cpp() does not have an entry for XRGB8888,
    # which causes the "bytes per pixel is invalid" WARN and silent
    # flip drop on every repaint.  ARGB8888 has the correct cpp=4
    # entry.  The DSI wire is rgb888 (per DTS) but the DPU internal
    # buffer format and the DSI output format are separate — the DPU
    # converts internally.
    # ---------------------------------------------------------------
    mkdir -p "$ALPINE/tmp/weston" 2>/dev/null || true
    cat > "$ALPINE/tmp/weston/weston.ini" << 'WESTONINI'
[core]
idle-time=0
require-input=false
xwayland=true

[shell]
background-color=0xff001a33
panel-position=none
locking=false

[output]
name=DSI-1
mode=current
pixel-format=argb8888
WESTONINI
    logok "wrote /tmp/weston/weston.ini (pixel-format=argb8888)"

    SEATD_SOCK_PATH="${SEATD_SOCK:-/run/seatd.sock}"
    [ -S "$SEATD_SOCK_PATH" ] \
        && logok "seatd socket confirmed at $SEATD_SOCK_PATH" \
        || logw  "seatd socket NOT found at $SEATD_SOCK_PATH — Weston will fail"

    [ -S "$ALPINE/run/seatd.sock" ] \
        && logok "seatd socket visible through chroot at $ALPINE/run/seatd.sock" \
        || logw  "seatd socket NOT visible through chroot at $ALPINE/run/seatd.sock"

    # ---------------------------------------------------------------
    # HOST-SIDE PRE-LAUNCH DIAGNOSTICS (runs outside chroot)
    # ---------------------------------------------------------------
    logsub "PRE-LAUNCH: DRM / framebuffer / kernel state"

    logi "--- dmesg: sprd/dpu/dsi/drm/panel (pre-launch) ---"
    dmesg | grep -iE 'sprd|dpu|dispc|drm|dsi|panel|flip|vblank|lcd|backlight|display|FC11339|mipi' \
          | tail -60 | tee -a "$LOGFILE" || true

    logi "--- /sys/class/drm/card0-DSI-1 ---"
    for f in status dpms modes enabled; do
        v=$(cat "/sys/class/drm/card0-DSI-1/$f" 2>/dev/null) \
            && logi "  card0-DSI-1/$f = $v" \
            || logi "  card0-DSI-1/$f = (not found)"
    done

    logi "--- /sys/class/graphics/fb0 ---"
    for f in virtual_size stride bits_per_pixel; do
        v=$(cat "/sys/class/graphics/fb0/$f" 2>/dev/null) \
            && logi "  fb0/$f = $v" \
            || logi "  fb0/$f = (not found)"
    done

    logi "--- DRI debug state (best-effort) ---"
    cat /sys/kernel/debug/dri/0/state 2>/dev/null | head -60 | tee -a "$LOGFILE" \
        || logi "  /sys/kernel/debug/dri/0/state not accessible"

    logi "--- DRM clients ---"
    cat /sys/kernel/debug/dri/0/clients 2>/dev/null | tee -a "$LOGFILE" \
        || logi "  /sys/kernel/debug/dri/0/clients not accessible"

    # ---------------------------------------------------------------
    # HOST-SIDE: DRM plane format advertisement
    # Shows which DRM_FORMAT_* codes the Sprd DPU driver actually
    # advertises for each plane.  If DRM_FORMAT_ARGB8888 is not in
    # this list, pixel-format=argb8888 will also fail and we need
    # to use fbdev-backend instead.
    # ---------------------------------------------------------------
    logi "--- DRM plane format advertisement (host sysfs) ---"
    for plane_dir in /sys/class/drm/card0/device/drm/card0/plane*/; do
        [ -d "$plane_dir" ] || continue
        plane_name=$(basename "$plane_dir")
        formats_file="$plane_dir/formats"
        if [ -r "$formats_file" ]; then
            logi "  $plane_name formats: $(cat "$formats_file" 2>/dev/null | tr '\n' ' ')"
        else
            logi "  $plane_name: no formats sysfs entry"
        fi
    done
    # Also try debugfs path used on some 4.14 vendor kernels
    for plane_dir in /sys/kernel/debug/dri/0/plane*/; do
        [ -d "$plane_dir" ] || continue
