Step 4
MIRROR="https://dl-cdn.alpinelinux.org/alpine"

ARCH="aarch64"

APK_STATIC_PKG=$(curl -fsSL "${MIRROR}/edge/main/${ARCH}/" \
  | grep -o 'apk-tools-static-[0-9][^"]*\.apk' | head -1)
curl -fsSL "${MIRROR}/edge/main/${ARCH}/${APK_STATIC_PKG}" -o /tmp/apk-tools-static.apk
tar -xzf /tmp/apk-tools-static.apk -C /tmp sbin/apk.static
chmod +x /tmp/sbin/apk.static


sudo mkdir -p mnt/etc/apk/keys
tar -tf /tmp/apk-tools-static.apk | grep '\.pub' | while read k; do
  tar -xzf /tmp/apk-tools-static.apk "$k" -O 2>/dev/null \
    | sudo tee "mnt/etc/apk/keys/$(basename "$k")" > /dev/null || true
done

sudo /tmp/sbin/apk.static \
  --arch "${ARCH}" \
  -X "${MIRROR}/edge/main" \
  -X "${MIRROR}/edge/community" \
  --keys-dir mnt/etc/apk/keys \
  -U --allow-untrusted \
  --root mnt \
  --initdb \
  add alpine-base

sudo cp /usr/bin/qemu-aarch64-static mnt/usr/bin/

sudo tee mnt/etc/resolv.conf > /dev/null <<'EOF'
nameserver 1.1.1.1
nameserver 8.8.8.8
EOF

sudo mount -t proc   proc   mnt/proc

sudo mount -t sysfs  sysfs  mnt/sys

sudo mount --bind    /dev   mnt/dev

sudo mount -t devpts devpts mnt/dev/pts

extract_appimage() {
  local src="$1" dest="$2"
  echo "Extracting: $(basename $src) -> $dest"
  chmod +x "$src"
  rm -rf "$dest"
  local work; work=$(mktemp -d /tmp/xwork_XXXXXX)
  cd "$work"
  /usr/bin/qemu-aarch64-static "$src" --appimage-extract >/dev/null 2>&1 || true
  local extracted=""
  [ -d "$work/squashfs-root" ] && extracted="$work/squashfs-root"
  [ -L "$work/squashfs-root" ] && extracted=$(readlink -f "$work/squashfs-root")
  [ -z "$extracted" ] && [ -d "$work/AppDir" ] && extracted="$work/AppDir"
  [ -z "$extracted" ] && extracted=$(find "$work" -maxdepth 1 -mindepth 1 -type d | head -1)
  if [ -z "$extracted" ] || [ ! -e "$extracted" ]; then
    echo "  ERROR: extraction failed"; ls "$work"
    cd /tmp; rm -rf "$work"; return 1
  fi
  mkdir -p "$dest"
  cp -a "$extracted"/. "$dest"/
  cd /tmp && rm -rf "$work"
  echo "  OK: $dest ($(find "$dest" -maxdepth 1 -mindepth 1 | wc -l) items)"
}

win_extract() {
  local zip="$1" dest="$2"
  [ -d "$dest" ] && [ "$(find $dest -maxdepth 1 | wc -l)" -gt 1 ] && \
    echo "  SKIP: $dest already populated" && return
  rm -rf "$dest"; mkdir -p "$dest"
  unzip -o -q "$zip" -d "$dest" || true
  local subdirs; subdirs=$(find "$dest" -maxdepth 1 -mindepth 1 -type d | wc -l)
  if [ "$subdirs" -eq 1 ]; then
    local sub; sub=$(find "$dest" -maxdepth 1 -mindepth 1 -type d)
    mv "$sub"/* "$dest"/ 2>/dev/null || true; rmdir "$sub" 2>/dev/null || true
  fi
  echo "  $dest: $(find $dest -maxdepth 1 | wc -l) items"
}
