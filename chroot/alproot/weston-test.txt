#!/system/bin/sh
# =====================================================
# SPREADTRUM/MALI WAYLAND LAUNCHER (TTY OPTION FIX)
# =====================================================

DEBIAN_ROOT="/mnt/debian"
BUSYBOX="/system/bin/busybox"
LOGFILE="/sdcard/wayland_drm.log"

# Clean log
echo "=== WAYLAND DRM LAUNCHER ===" > "$LOGFILE"
echo "Started: $(date '+%Y-%m-%d %H:%M:%S')" >> "$LOGFILE" 2>/dev/null
sync

if [ ! -w "$LOGFILE" ]; then
    LOGFILE="/tmp/wayland_drm.log"
    echo "=== WAYLAND DRM LAUNCHER ===" > "$LOGFILE"
    echo "Started: $(date '+%Y-%m-%d %H:%M:%S')" >> "$LOGFILE"
fi

log() {
    local msg="[$(date '+%H:%M:%S')] $*"
    echo "$msg" | tee -a "$LOGFILE"
    sync
}

log_section() {
    echo "" | tee -a "$LOGFILE"
    log "========================================="
    log "=== $* ==="
    log "========================================="
}

log_info() { log "  ℹ $*"; }
log_success() { log "  ✓ $*"; }

# === INITIAL WAIT ===
log_section "INITIALIZATION"
log_info "Waiting 30 seconds for system to stabilize..."
COUNT=0
while [ $COUNT -lt 30 ]; do
    $BUSYBOX sleep 1
    COUNT=$((COUNT + 1))
    if [ $((COUNT % 5)) -eq 0 ]; then
        log_info "  $COUNT/30 seconds..."
    fi
done
log_success "System ready"
log ""

# === 1. POWER MANAGEMENT ===
log "--- Waking up Display Pipeline ---"
if [ -e /sys/class/graphics/fb0/blank ]; then
    log "Unblanking fb0..."
    echo 0 > /sys/class/graphics/fb0/blank
fi
for bl in /sys/class/backlight/*; do
    if [ -e "$bl/brightness" ]; then
        max=$(cat "$bl/max_brightness")
        log "Setting $bl to $max"
        echo "$max" > "$bl/brightness"
    fi
done

# === 2. MOUNT & PERMISSIONS ===
# Ensure /dev/input is mounted so Weston sees touch/buttons
if [ ! -c "$DEBIAN_ROOT/dev/input/event0" ]; then
    log "Mounting /dev..."
    mount -o bind /dev "$DEBIAN_ROOT/dev"
    mount -o bind /sys "$DEBIAN_ROOT/sys"
    mount -o bind /proc "$DEBIAN_ROOT/proc"
    mount -t tmpfs tmpfs "$DEBIAN_ROOT/run"
fi

# Fix permissions for ALL input devices and card0 and mali0
chmod 666 "$DEBIAN_ROOT/dev/dri/card0"
chmod 666 "$DEBIAN_ROOT/dev/mali0" 2>/dev/null
chmod 666 "$DEBIAN_ROOT/dev/ion" 2>/dev/null
chmod -R 666 "$DEBIAN_ROOT/dev/input/" 2>/dev/null

# === 3. LAUNCH WESTON ===
log "--- Starting Weston ---"

chroot "$DEBIAN_ROOT" /bin/bash <<'EOF_CHROOT' 2>&1 | tee -a "$LOGFILE"
export PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
export LD_LIBRARY_PATH=/usr/lib/aarch64-linux-gnu:/usr/lib:/lib
export XDG_RUNTIME_DIR=/tmp/runtime-root

mkdir -p -m 0700 "$XDG_RUNTIME_DIR"
rm -f "$XDG_RUNTIME_DIR"/*

export LIBSEAT_BACKEND=builtin
export SEATD_VTBOUND=0
export LIBGL_ALWAYS_SOFTWARE=1
export EGL_PLATFORM=surfaceless

# Create config file
mkdir -p /root/.config
cat > /root/.config/weston.ini <<INI
[core]
backend=drm-backend.so
shell=kiosk-shell.so
use-pixman=true
idle-time=0
require-input=false

[shell]
background-color=0xff0000ff
panel-position=none
locking=false

[output]
name=DSI-1
mode=current
transform=rotate-90
INI

export WESTON_DRM_DEBUG=1

# Run Weston WITHOUT --tty option
# We also use 'timeout' to stop it after 15 seconds if it hangs successfully (so you get logs)
# or remove 'timeout 15s' if you want it to run forever until you reboot.
echo "Launching Weston..."
weston --config=/root/.config/weston.ini --log=/tmp/weston.log --debug &
PID=$!

sleep 10
if ps -p $PID > /dev/null; then
   echo "Weston is still running (SUCCESS!)"
   kill $PID
else
   echo "Weston crashed/exited"
fi
wait $PID
EXIT_CODE=$?

echo ""
echo "--- WESTON LOG (/tmp/weston.log) ---"
if [ -f /tmp/weston.log ]; then
    cat /tmp/weston.log
else
    echo "ERROR: Log file not found!"
fi
echo "--- END WESTON LOG ---"
echo ""

echo "Weston exit: $EXIT_CODE"
EOF_CHROOT

log "--- DONE ---"

exec /system/bin/realrecovery "$@"
"
