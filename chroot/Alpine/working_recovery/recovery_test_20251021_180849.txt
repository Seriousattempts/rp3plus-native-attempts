#!/system/bin/sh
# Alpine Linux Complete Hardware Testing Script
# Clean install with manual /dev/ttyBT0 and /dev/i2c-* devices

# === PARTITION CONFIGURATION ===
SUPER_PARTITION="/dev/block/mmcblk0p50"
DISTRO_PARTITION="/dev/block/mmcblk0p77"
DISTRO_MOUNT="/mnt/distro"
DISTRO_ROOT=""
BOOT_PARTITION="/dev/block/mmcblk1p1"
SOCKO_PARTITION="/dev/block/mmcblk0p52"
SD_MOUNT=""
TEMP_DIR=""
LOGFILE="/tmp/alpine_hardware_test.log"

# === LOGGING SETUP ===
> "$LOGFILE"
log() {
    echo "[$(date '+%H:%M:%S')] $*" | tee -a "$LOGFILE"
}

log "=== Alpine Linux Complete Hardware Testing Script ==="
log "Device: Retroid Pocket 3 Plus (UMS512 / Mali-G52)"
log "Kernel: $(uname -a)"

# === BOOT PARTITION MOUNT ===
log ""
log "=== BOOT PARTITION ==="
if grep -q "$BOOT_PARTITION" /proc/mounts 2>/dev/null; then
    SD_MOUNT=$(grep "$BOOT_PARTITION" /proc/mounts | awk '{print $2}' | head -1)
    log "✓ Already mounted: $SD_MOUNT"
else
    for mp in /sdcard /boot /mnt/boot; do
        mkdir -p "$mp" 2>/dev/null
        if mount -t exfat $BOOT_PARTITION "$mp" 2>/dev/null; then
            SD_MOUNT="$mp"
            log "✓ Mounted: $SD_MOUNT"
            break
        fi
    done
    if [ -z "$SD_MOUNT" ]; then
        SD_MOUNT="/tmp/boot"
        mkdir -p "$SD_MOUNT"
        log "⚠ Using fallback: $SD_MOUNT"
    fi
fi

mkdir -p "$SD_MOUNT/logs" 2>/dev/null

if [ "$SD_MOUNT" != "/tmp/boot" ] && [ -w "$SD_MOUNT/logs" ]; then
    NEW_LOG="${SD_MOUNT}/logs/alpine_complete_test_$(date +%Y%m%d_%H%M%S).log"
    cat "$LOGFILE" > "$NEW_LOG" 2>/dev/null && LOGFILE="$NEW_LOG"
    log "✓ Log relocated: $LOGFILE"
fi

# === WAIT FOR BLOCK DEVICES ===
log ""
log "=== WAITING FOR BLOCK DEVICES ==="
WAIT=0
while [ ! -b $SUPER_PARTITION ] && [ $WAIT -lt 50 ]; do
    sleep 0.1
    WAIT=$((WAIT + 1))
done

if [ ! -b $SUPER_PARTITION ]; then
    log "ERROR: Super partition not found"
    exec /system/bin/realrecovery "$@"
fi
log "✓ Super partition found"

WAIT=0
while [ ! -b $DISTRO_PARTITION ] && [ $WAIT -lt 50 ]; do
    sleep 0.1
    WAIT=$((WAIT + 1))
done

if [ ! -b $DISTRO_PARTITION ]; then
    log "ERROR: Distro partition not found"
    exec /system/bin/realrecovery "$@"
fi
log "✓ Distro partition found"

# === MOUNT DISTRO PARTITION ===
log ""
log "=== DISTRO PARTITION MOUNT ==="
mkdir -p $DISTRO_MOUNT

if mount $DISTRO_PARTITION $DISTRO_MOUNT 2>&1 | tee -a "$LOGFILE"; then
    log "✓ Mounted: $DISTRO_MOUNT"
else
    log "ERROR: Mount failed"
    exec /system/bin/realrecovery "$@"
fi

TEMP_DIR="$DISTRO_MOUNT/super_temp"
mkdir -p "$TEMP_DIR" 2>/dev/null

# === FIND ALPINE ROOTFS ===
log ""
log "=== ROOTFS DISCOVERY ==="
DISTRO_ROOT=$(find "$DISTRO_MOUNT" -type d -name "alpine" -path "*/chroot-distro/alpine" 2>/dev/null | head -1)

if [ -z "$DISTRO_ROOT" ]; then
    log "ERROR: Alpine rootfs not found"
    umount $DISTRO_MOUNT 2>/dev/null
    exec /system/bin/realrecovery "$@"
fi

log "✓ Rootfs: $DISTRO_ROOT"

# === MOUNT FUNCTION ===
mount_partition() {
    local name=$1
    local start_sector=$2
    local num_sectors=$3
    local mount_point="$DISTRO_MOUNT/$name"
    local img_file="${TEMP_DIR}/${name}.img"
    
    dd if=$SUPER_PARTITION of=$img_file bs=4096 skip=$((start_sector / 8)) count=$((num_sectors / 8)) 2>&1 | tee -a "$LOGFILE"
    
    if [ -f "$img_file" ]; then
        local loop_dev=$(losetup -f 2>/dev/null)
        if [ -n "$loop_dev" ]; then
            losetup $loop_dev $img_file 2>&1 | tee -a "$LOGFILE"
            mount -t ext4 -o rw $loop_dev $mount_point 2>&1 | tee -a "$LOGFILE"
            return $?
        fi
    fi
    return 1
}

# === MOUNT PARTITIONS ===
log ""
log "=== MOUNTING PARTITIONS ==="
mkdir -p $DISTRO_MOUNT/{system,vendor,system_ext,product} 2>/dev/null

mount_partition "system" 2048 2692280
mount_partition "system_ext" 2695168 741136
mount_partition "vendor" 3436544 3700976
mount_partition "product" 7139328 1190008

# === CHROOT SETUP ===
log ""
log "=== CHROOT SETUP ==="

mkdir -p "$DISTRO_ROOT/etc"
cat > "$DISTRO_ROOT/etc/ld.so.conf" << 'EOF'
/lib
/usr/lib
/system_ext/lib
/system_ext/lib64
/system/lib
/system/lib64
/system/lib/hw
/system/lib/drm
/system/lib/bootstrap
/system/lib64/hw
/system/lib64/drm
/system/lib64/bootstrap
/system/lib64/extractors
/vendor/lib
/vendor/lib/egl
/vendor/lib/hw
/vendor/lib/mediacas
/vendor/lib/mediadrm
/vendor/lib/modules
/vendor/lib/mnpidevice
/vendor/lib/soundfx
/vendor/lib64
/vendor/lib64/egl
/vendor/lib64/hw
/vendor/lib64/mediacas
/vendor/lib64/mediadrm
/vendor/lib64/mnpidevice
/vendor/lib64/soundfx
EOF

mkdir -p "$DISTRO_ROOT/dev" 2>/dev/null
mount --bind /dev "$DISTRO_ROOT/dev" 2>/dev/null

mkdir -p "$DISTRO_ROOT/dev/pts" 2>/dev/null
mount -t devpts devpts "$DISTRO_ROOT/dev/pts" -o noexec,nosuid,gid=5,mode=620 2>/dev/null

mount -t proc proc "$DISTRO_ROOT/proc" 2>/dev/null
mount -t sysfs sysfs "$DISTRO_ROOT/sys" 2>/dev/null

# === BIND MOUNT PARTITIONS ===
for partition in system system_ext vendor product; do
    if mount | grep -q "$DISTRO_MOUNT/$partition"; then
        mkdir -p "$DISTRO_ROOT/$partition" 2>/dev/null
        mount --bind "$DISTRO_MOUNT/$partition" "$DISTRO_ROOT/$partition" 2>/dev/null
    fi
done

# === MOUNT SOCKO ===
SOCKO_MOUNT="$DISTRO_ROOT/mnt/vendor/socko"
mkdir -p "$SOCKO_MOUNT" 2>/dev/null

if [ -b "$SOCKO_PARTITION" ]; then
    mount -t ext4 -o ro,noatime,nosuid,nodev "$SOCKO_PARTITION" "$SOCKO_MOUNT" 2>&1 | tee -a "$LOGFILE"
fi

mkdir -p "$DISTRO_ROOT/sdcard" 2>/dev/null
mount --bind "$SD_MOUNT" "$DISTRO_ROOT/sdcard" 2>/dev/null

cat > "$DISTRO_ROOT/etc/resolv.conf" << 'EOF'
nameserver 8.8.8.8
nameserver 1.1.1.1
EOF

# === USER SETUP ===
log ""
log "=== USER SETUP ==="

/system/bin/busybox chroot "$DISTRO_ROOT" /bin/ash -c "
    chown -R p3plus:p3plus /home/p3plus
" 2>&1 | tee -a "$LOGFILE"

log "✓ User configured"

# === CREATE DROID LIBRARY DIRECTORY ===
log ""
log "=== CREATING DROID LIBRARY DIRECTORY ==="

mkdir -p "$DISTRO_ROOT/droid/lib64" 2>/dev/null

# Copy Android graphics libraries to test directory
if [ -f "$DISTRO_MOUNT/vendor/lib64/hw/android.hardware.graphics.mapper@4.0-impl-arm.so" ]; then
    cp "$DISTRO_MOUNT/vendor/lib64/hw/android.hardware.graphics.mapper@4.0-impl-arm.so" "$DISTRO_ROOT/droid/lib64/" 2>/dev/null
fi

if [ -f "$DISTRO_MOUNT/vendor/lib64/hw/android.hardware.renderscript@1.0-impl.so" ]; then
    cp "$DISTRO_MOUNT/vendor/lib64/hw/android.hardware.renderscript@1.0-impl.so" "$DISTRO_ROOT/droid/lib64/" 2>/dev/null
fi

if [ -f "$DISTRO_MOUNT/vendor/lib64/hw/gralloc.default.so" ]; then
    cp "$DISTRO_MOUNT/vendor/lib64/hw/gralloc.default.so" "$DISTRO_ROOT/droid/lib64/" 2>/dev/null
fi

if [ -f "$DISTRO_MOUNT/vendor/lib64/egl/libGLES_mali.so" ]; then
    cp "$DISTRO_MOUNT/vendor/lib64/egl/libGLES_mali.so" "$DISTRO_ROOT/droid/lib64/" 2>/dev/null
fi

if [ -f "$DISTRO_MOUNT/vendor/lib64/libOpenCL.so" ]; then
    cp "$DISTRO_MOUNT/vendor/lib64/libOpenCL.so" "$DISTRO_ROOT/droid/lib64/" 2>/dev/null
fi

log "✓ Droid libraries copied"

# === RUN COMPREHENSIVE HARDWARE TEST ===
log ""
log "=== RUNNING COMPREHENSIVE HARDWARE TEST ==="

TEST_OUTPUT="/sdcard/logs/complete_test_$(date +%Y%m%d_%H%M%S).log"

/system/bin/busybox chroot "$DISTRO_ROOT" /bin/ash -c "
    export PATH='/bin:/usr/bin:/sbin:/usr/sbin:/system/bin:/vendor/bin'
    export LD_LIBRARY_PATH='/lib:/usr/lib:/system/lib64:/system/lib:/system_ext/lib64:/system_ext/lib:/vendor/lib64:/vendor/lib:/vendor/lib64/egl:/vendor/lib64/hw'
    export HOME='/home/p3plus'
    export USER='root'
    export TERM='linux'
    export XDG_RUNTIME_DIR=/tmp/runtime-0
    
    mkdir -p \$XDG_RUNTIME_DIR
    chmod 700 \$XDG_RUNTIME_DIR
    
    exec > '$TEST_OUTPUT' 2>&1
    
    echo '======================================================================='
    echo 'ALPINE LINUX COMPLETE HARDWARE TEST'
    echo 'Device: Retroid Pocket 3 Plus (UMS512 / Mali-G52)'
    echo 'Date: '\$(date)
    echo 'Kernel: '\$(uname -a)
    echo '======================================================================='
    echo ''

    # === LOAD MODULES ===
    echo '=== LOADING KERNEL MODULES ==='
    SOCKO='/mnt/vendor/socko'
    for mod in mali_gondul.ko sprd_vdsp.ko mmdvfs.ko sprd_cpp.ko sprd_sensor.ko sprd_flash_drv.ko sprdbt_tty.ko sprd_fm.ko sprdwl_ng.ko; do
        [ -f \"\$SOCKO/\$mod\" ] && insmod \"\$SOCKO/\$mod\" 2>&1 && echo \"✓ \$mod\" && sleep 1
    done
    echo ''
    
    pkill mdevd 2>/dev/null; sleep 1; mdevd & sleep 3; mdev -s
    echo ''

    # === WIFI TEST ===
    echo '======================================================================='
    echo 'TEST: WIFI'
    echo '======================================================================='
    echo ''
    
    sleep 5
    WIFI_IF=\$(ip link show 2>/dev/null | grep -o 'wlan[0-9]*' | head -1)
    
    if [ -n \"\$WIFI_IF\" ]; then
        echo \"✓ WiFi interface: \$WIFI_IF\"
        ip link set \"\$WIFI_IF\" up
        sleep 3
        ip addr show \"\$WIFI_IF\"
        echo ''
        
        if command -v iw >/dev/null 2>&1; then
            echo 'Scanning networks (abbreviated)...'
            iw dev \"\$WIFI_IF\" scan 2>&1 | grep -E 'BSS|SSID|signal' | head -40
            echo ''
            NETWORKS=\$(iw dev \"\$WIFI_IF\" scan 2>/dev/null | grep -c 'SSID:')
            echo \"Networks found: \$NETWORKS\"
        fi
        echo ''
        echo '✓ WIFI TEST PASSED'
    else
        echo '✗ No WiFi interface'
        echo '✗ WIFI TEST FAILED'
    fi
    echo ''

    # === ENHANCED MODEM TEST ===
    echo '======================================================================='
    echo 'TEST: MODEM & DIGITAL SIGNAL PROCESSOR'
    echo '======================================================================='
    echo ''
    
    echo '--- Starting D-Bus (Required for ModemManager) ---'
    if ! pgrep -x dbus-daemon >/dev/null; then
        mkdir -p /run/dbus
        dbus-daemon --system --fork 2>&1
        sleep 2
        echo '✓ D-Bus started'
    else
        echo '✓ D-Bus already running'
    fi
    echo ''
    
    echo '--- Modem Kernel Devices ---'
    ls -la /dev/tty* | grep -E 'ACM|USB|LTE|modem' || echo 'No modem TTY devices'
    ls -la /dev/smd* 2>/dev/null || echo 'No SMD devices'
    ls -la /dev/esoc* 2>/dev/null || echo 'No ESOC devices'
    echo ''
    
    echo '--- QMI Devices (Qualcomm MSM Interface) ---'
    ls -la /dev/cdc-wdm* /dev/qmi* 2>/dev/null || echo 'No QMI devices'
    echo ''
    
    echo '--- MBIM Devices (Mobile Broadband Interface) ---'
    ls -la /dev/cdc-wdm* /dev/mbim* 2>/dev/null || echo 'No MBIM devices'
    echo ''
    
    echo '--- USB Modem Detection ---'
    if command -v lsusb >/dev/null 2>&1; then
        lsusb 2>&1 | grep -i -E 'modem|qualcomm|sierra|huawei|zte|mediatek'
    else
        ls /sys/bus/usb/devices/*/product 2>/dev/null | xargs cat | grep -i modem
    fi
    echo ''
    
    echo '--- PCI/PCIe Modem Detection ---'
    if command -v lspci >/dev/null 2>&1; then
        lspci 2>&1 | grep -i -E 'modem|wwan|lte|baseband'
    else
        ls /sys/bus/pci/devices/*/class 2>/dev/null | xargs grep -l 0x0280 | xargs dirname | xargs -I {} cat {}/uevent
    fi
    echo ''
    
    echo '--- SPRD Modem Devices (Platform-specific) ---'
    ls -la /dev/stty* /dev/slog* /dev/spipe* 2>/dev/null || echo 'No SPRD modem devices'
    ls -la /sys/class/sprd_* 2>/dev/null || echo 'No SPRD classes'
    echo ''
    
    echo '--- Digital Signal Processor (DSP) Devices ---'
    ls -la /dev/sprd_vdsp* /dev/dsp* /dev/adsp* 2>/dev/null || echo 'No DSP devices'
    echo ''
    
    echo '--- DSP Kernel Module ---'
    lsmod | grep -i vdsp
    echo ''
    
    echo '--- DSP Sysfs Information ---'
    if [ -d /sys/devices/platform/soc/soc:mm ]; then
        find /sys/devices/platform/soc/soc:mm -name '*vdsp*' -o -name '*dsp*' 2>/dev/null
        for dsp_dev in \$(find /sys/devices/platform/soc/soc:mm -name '*vdsp*' 2>/dev/null); do
            echo \"DSP Device: \$dsp_dev\"
            ls -la \"\$dsp_dev\"/ 2>/dev/null | head -10
            echo ''
        done
    fi
    
    echo '--- ModemManager Status ---'
    if command -v mmcli >/dev/null 2>&1; then
        echo 'Starting ModemManager (background)...'
        ModemManager --debug &
        MM_PID=\$!
        echo \"ModemManager PID: \$MM_PID\"
        sleep 5
        
        echo ''
        echo 'Listing modems:'
        mmcli -L 2>&1
        echo ''
        
        MODEM=\$(mmcli -L 2>/dev/null | grep -o '/Modem/[0-9]*' | head -1)
        if [ -n \"\$MODEM\" ]; then
            echo \"✓ Found modem: \$MODEM\"
            echo ''
            mmcli -m \$MODEM 2>&1
            echo ''
            echo 'Modem capabilities:'
            mmcli -m \$MODEM --list-bearers 2>&1
        else
            echo '✗ No modems detected by ModemManager'
        fi
        
        kill \$MM_PID 2>/dev/null
        sleep 1
    else
        echo 'ModemManager not installed'
    fi
    echo ''
    
    echo '--- QMI Device Info (if available) ---'
    if command -v qmicli >/dev/null 2>&1; then
        for wdm in /dev/cdc-wdm*; do
            if [ -c \"\$wdm\" ]; then
                echo \"Testing \$wdm:\"
                timeout 5 qmicli -d \"\$wdm\" --dms-get-manufacturer 2>&1
                timeout 5 qmicli -d \"\$wdm\" --dms-get-model 2>&1
                timeout 5 qmicli -d \"\$wdm\" --dms-get-revision 2>&1
                echo ''
            fi
        done
    else
        echo 'qmicli not installed'
    fi
    echo ''
    
    echo '--- MBIM Device Info (if available) ---'
    if command -v mbimcli >/dev/null 2>&1; then
        for wdm in /dev/cdc-wdm*; do
            if [ -c \"\$wdm\" ]; then
                echo \"Testing \$wdm:\"
                timeout 5 mbimcli -d \"\$wdm\" --query-device-caps 2>&1
                echo ''
            fi
        done
    else
        echo 'mbimcli not installed'
    fi
    echo ''
    
    echo '--- NetworkManager WWAN Status ---'
    if command -v nmcli >/dev/null 2>&1; then
        nmcli device status 2>&1
        echo ''
        nmcli connection show 2>&1 | grep -i 'gsm\|wwan\|mobile'
    else
        echo 'NetworkManager not installed'
    fi
    echo ''
    
    echo '--- Modem Kernel Messages ---'
    dmesg | grep -i -E 'modem|qmi|mbim|wwan|lte|gsm|umts|rild|ril' | tail -40
    echo ''
    
    echo '--- Modem Test Summary ---'
    echo \"QMI devices: \$(ls /dev/cdc-wdm* 2>/dev/null | wc -l)\"
    echo \"SPRD modem devices: \$(ls /dev/stty* /dev/slog* 2>/dev/null | wc -l)\"
    echo \"DSP devices: \$(ls /dev/*dsp* 2>/dev/null | wc -l)\"
    echo \"DSP module loaded: \$(lsmod | grep -c vdsp)\"
    echo ''
    
    echo '✓ MODEM TEST COMPLETED'
    echo ''

    # === CAMERA TEST (SYSFS ONLY - NO I2C BUS SCANNING) ===
    echo '======================================================================='
    echo 'TEST: CAMERA SUBSYSTEM (SYSFS DETECTION)'
    echo '======================================================================='
    echo ''
    
    echo '--- Camera Kernel Modules ---'
    lsmod | grep -E 'sprd.*sensor|sprd.*cpp|sprd.*flash'
    echo ''
    
    echo '--- Camera Device Nodes ---'
    ls -la /dev/sprd* 2>/dev/null || echo 'No SPRD devices'
    echo ''
    
    echo '--- I2C Device Nodes (0-5) ---'
    for i in 0 1 2 3 4 5; do
        if [ -c /dev/i2c-\$i ]; then
            echo \"✓ /dev/i2c-\$i exists\"
            ls -la /dev/i2c-\$i
        else
            echo \"✗ Missing: /dev/i2c-\$i\"
        fi
    done
    echo ''
    
    echo '--- Camera I2C Devices (Sysfs Detection) ---'
    echo 'Detected camera devices on I2C buses:'
    for cam_dev in 0-0020 0-0021 1-002a 1-006c; do
        if [ -d /sys/bus/i2c/devices/\$cam_dev ]; then
            echo ''
            echo \"✓ Device: \$cam_dev\"
            echo \"  Name:      \$(cat /sys/bus/i2c/devices/\$cam_dev/name 2>/dev/null || echo 'N/A')\"
            echo \"  Modalias:  \$(cat /sys/bus/i2c/devices/\$cam_dev/modalias 2>/dev/null || echo 'N/A')\"
            
            # Check if driver is bound
            if [ -e /sys/bus/i2c/devices/\$cam_dev/driver ]; then
                DRIVER=\$(ls -l /sys/bus/i2c/devices/\$cam_dev/driver 2>/dev/null | awk '{print \$NF}' | xargs basename)
                echo \"  Driver:    \$DRIVER\"
            else
                echo \"  Driver:    NOT BOUND\"
            fi
            
            # List device attributes
            echo \"  Attributes:\"
            ls /sys/bus/i2c/devices/\$cam_dev/ 2>/dev/null | grep -v 'power\|subsystem\|of_node\|driver\|uevent' | head -10 | sed 's/^/    /'
        else
            echo \"✗ Device \$cam_dev not detected\"
        fi
    done
    echo ''
    
    echo '--- Camera Sensor Identification ---'
    echo 'Main Camera @0x20 (i2c-0):'
    if [ -d /sys/bus/i2c/devices/0-0020 ]; then
        cat /sys/bus/i2c/devices/0-0020/name 2>/dev/null || echo '  Name not available'
        cat /sys/bus/i2c/devices/0-0020/modalias 2>/dev/null || echo '  Modalias not available'
    else
        echo '  NOT DETECTED'
    fi
    echo ''
    
    echo 'Deputy Camera @0x21 (i2c-0):'
    if [ -d /sys/bus/i2c/devices/0-0021 ]; then
        cat /sys/bus/i2c/devices/0-0021/name 2>/dev/null || echo '  Name not available'
        cat /sys/bus/i2c/devices/0-0021/modalias 2>/dev/null || echo '  Modalias not available'
    else
        echo '  NOT DETECTED'
    fi
    echo ''
    
    echo 'Main2 Camera @0x2A (i2c-1):'
    if [ -d /sys/bus/i2c/devices/1-002a ]; then
        cat /sys/bus/i2c/devices/1-002a/name 2>/dev/null || echo '  Name not available'
        cat /sys/bus/i2c/devices/1-002a/modalias 2>/dev/null || echo '  Modalias not available'
    else
        echo '  NOT DETECTED'
    fi
    echo ''
    
    echo 'Sub Camera @0x6C (i2c-1):'
    if [ -d /sys/bus/i2c/devices/1-006c ]; then
        cat /sys/bus/i2c/devices/1-006c/name 2>/dev/null || echo '  Name not available'
        cat /sys/bus/i2c/devices/1-006c/modalias 2>/dev/null || echo '  Modalias not available'
    else
        echo '  NOT DETECTED'
    fi
    echo ''
    
    echo '--- Voltage Regulators (Camera Power) ---'
    echo 'Camera-related regulators:'
    for reg in 4 7 8 11 12; do
        if [ -d /sys/class/regulator/regulator.\$reg ]; then
            NAME=\$(cat /sys/class/regulator/regulator.\$reg/name 2>/dev/null)
            STATE=\$(cat /sys/class/regulator/regulator.\$reg/state 2>/dev/null)
            USERS=\$(cat /sys/class/regulator/regulator.\$reg/num_users 2>/dev/null)
            VOLT=\$(cat /sys/class/regulator/regulator.\$reg/microvolts 2>/dev/null)
            echo \"  regulator.\$reg (\$NAME):\"
            echo \"    State: \$STATE\"
            echo \"    Users: \$USERS\"
            [ -n \"\$VOLT\" ] && echo \"    Voltage: \$((VOLT / 1000)) mV\"
        fi
    done
    echo ''
    
    echo '--- V4L2 Video Devices ---'
    if [ -d /sys/class/video4linux ]; then
        echo 'V4L2 devices found:'
        for v4l_dev in /sys/class/video4linux/video*; do
            if [ -d \"\$v4l_dev\" ]; then
                DEV_NAME=\$(basename \"\$v4l_dev\")
                V4L_NAME=\$(cat \"\$v4l_dev/name\" 2>/dev/null || echo 'N/A')
                echo \"  \$DEV_NAME: \$V4L_NAME\"
            fi
        done
    else
        echo 'No V4L2 devices detected'
    fi
    echo ''
    
    if command -v v4l2-ctl >/dev/null 2>&1; then
        echo 'V4L2 device enumeration:'
        v4l2-ctl --list-devices 2>&1 || echo 'No V4L2 devices'
        echo ''
        
        # List formats for each video device
        for vid in /dev/video*; do
            if [ -c \"\$vid\" ]; then
                echo \"Formats for \$vid:\"
                v4l2-ctl -d \"\$vid\" --list-formats-ext 2>&1 | head -20
                echo ''
            fi
        done
    fi
    
    echo '--- Media Controller Devices ---'
    if [ -d /sys/class/media ]; then
        for media_dev in /sys/class/media/media*; do
            if [ -d \"\$media_dev\" ]; then
                echo \"Device: \$(basename \$media_dev)\"
                cat \"\$media_dev/model\" 2>/dev/null || echo '  Model: N/A'
            fi
        done
    else
        echo 'No media controller devices'
    fi
    echo ''
    
    echo '--- Camera Kernel Messages ---'
    dmesg | grep -i -E 'sprd.*sensor|camera|dcam|isp|0-0020|1-006c|sensor.*sub|sensor.*main' | tail -40
    echo ''
    
    echo '--- Camera Test Summary ---'
    echo \"Kernel modules loaded: \$(lsmod | grep -c sprd_sensor)\"
    echo \"I2C camera devices (sysfs): \$(ls -d /sys/bus/i2c/devices/{0-0020,0-0021,1-002a,1-006c} 2>/dev/null | wc -l)/4\"
    echo \"Camera regulators enabled: \$(cat /sys/class/regulator/regulator.{4,7,8,11,12}/state 2>/dev/null | grep -c enabled)/5\"
    echo \"V4L2 devices: \$(ls /dev/video* 2>/dev/null | wc -l)\"
    echo \"Media devices: \$(ls /dev/media* 2>/dev/null | wc -l)\"
    
    echo ''
    echo '✓ CAMERA TEST COMPLETED (Sysfs detection only - no I2C bus probing)'
    echo ''

    # === COMPREHENSIVE GRAPHICS TEST ===
    echo '======================================================================='
    echo 'TEST: GRAPHICS SUBSYSTEM (COMPLETE)'
    echo '======================================================================='
    echo ''
    
    echo '--- Video Group Members ---'
    getent group video
    echo ''
    
    echo '--- All Graphics Device Nodes ---'
    echo 'DRM devices:'

    echo 'Android libraries (/droid/lib64):'

EOFVK
        
        if command -v vulkaninfo >/dev/null 2>&1; then
            echo \"Using: \$VK_ICD_FILENAMES\"
            vulkaninfo --summary 2>&1 | head -100
        fi

    echo '======================================================================='
    echo 'TEST: SENSORS (IIO + LM-SENSORS - SYSFS ONLY)'
    echo '======================================================================='
    echo ''
    
    echo '--- IIO Devices ---'
    ls -la /dev/iio:device* 2>/dev/null
    echo ''
    
    for dev in /sys/bus/iio/devices/iio:device*; do

            echo \"  Path: \$dev\"

            echo \"  Channels:\"
            ls \"\$dev\" | grep 'in_' | head -10 | sed 's/^/    /'

            for attr in in_accel_x_raw in_accel_y_raw in_accel_z_raw in_voltage_raw; do

                    VAL=\$(cat \"\$dev/\$attr\" 2>/dev/null || echo 'N/A')
                    echo \"    \$attr: \$VAL\"

            done
            echo ''
        fi
    done
    
    echo '--- IIO Sensor Proxy ---'
    if command -v iio-sensor-proxy >/dev/null 2>&1; then
        iio-sensor-proxy --version 2>&1
    else
        echo 'iio-sensor-proxy not installed'
    fi
    echo ''
    
    echo '--- LM-Sensors Detection (SKIPPED - Causes hangs on I2C buses) ---'
    echo 'sensors-detect would scan I2C buses which causes system hangs'
    echo 'Skipping automatic detection...'
    echo ''
    
    echo '--- Sensor Readings (lm-sensors) ---'

        sensors 2>&1
    else
        echo 'lm-sensors not installed or no sensors configured'

    echo ''

                for dev in \$DEVICES; do
                    ADDR=\$(basename \"\$dev\" | cut -d'-' -f2)

                    MODALIAS=\$(cat \"\$dev/modalias\" 2>/dev/null || echo 'N/A')

            else
                echo \"  No devices detected on this bus\"
            fi
        fi
        echo ''
    done
    
    echo '--- Hardware Monitoring Devices (hwmon) ---'
    if [ -d /sys/class/hwmon ]; then
        for hwmon in /sys/class/hwmon/hwmon*; do

                    fi

    echo \"hwmon devices: \$(ls -d /sys/class/hwmon/hwmon* 2>/dev/null | wc -l)\"
    echo \"Thermal zones: \$(ls -d /sys/class/thermal/thermal_zone* 2>/dev/null | wc -l)\"
    echo ''
    echo '⚠ I2C bus scanning skipped to prevent system hangs'
